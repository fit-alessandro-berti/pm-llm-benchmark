Assuming every order completes all activities exactly once, with no optional steps or rework, and follows the stated linear sequence:

```python
activity_order = (
    "RO",   # Receive Order
    "CA",   # Check Availability
    "RI",   # Reserve Items
    "PI",   # Pick Items
    "QC",   # Quality Check
    "PO",   # Pack Order
    "GSL",  # Generate Shipping Label
    "D",    # Dispatch
    "IC",   # Invoice Customer
    "RP",   # Record Payment
)

log_skeleton = {
    # All activities have the same occurrence count: exactly once.
    "equivalence": {
        (a, b)
        for a in activity_order
        for b in activity_order
        if a != b
    },

    # Tuple orientation: (activity, activity_that_must_be_before_it).
    "always_before": {
        (later, earlier)
        for i, earlier in enumerate(activity_order)
        for later in activity_order[i + 1:]
    },

    # Tuple orientation: (activity, activity_that_must_be_after_it).
    "always_after": {
        (earlier, later)
        for i, earlier in enumerate(activity_order)
        for later in activity_order[i + 1:]
    },

    # No activities are mutually exclusive in this workflow.
    "never_together": set(),

    # Every activity occurs exactly once per order.
    "activ_freq": {
        activity: {1}
        for activity in activity_order
    },

    # Immediate sequence constraints.
    "directly_follows": set(zip(activity_order, activity_order[1:]))
}
```
