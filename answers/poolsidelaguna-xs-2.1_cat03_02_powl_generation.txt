from pm4py.objects.powl.obj import StrictPartialOrder, OperatorPOWL, Transition, SilentTransition
from pm4py.objects.process_tree.obj import Operator

# Define all transition nodes
SR = Transition("Service Request")
CS = Transition("Contract Setup")
DA = Transition("Datacenter Allocation")
ACI = Transition("Access Control Initialization")
BNC = Transition("Base Network Configuration")
D_R = Transition("Debug & Reconfigure")
CD = Transition("Containerized Deployment")
MD = Transition("Monolithic Deployment")
QA = Transition("Quality Assurance")
SA = Transition("Security Audit")
GLA = Transition("Go-Live Approval")

# Define the loop node (BNC and D&R)
loop_node = OperatorPOWL(operator=Operator.LOOP, children=[BNC, D_R])

# Define the exclusive choice node (CD and MD)
choice_node = OperatorPOWL(operator=Operator.XOR, children=[CD, MD])

# Define the partial order for DA and ACI (concurrent)
po_da_aci = StrictPartialOrder(nodes=[DA, ACI])

# Define the partial order for QA and SA (concurrent)
po_qa_sa = StrictPartialOrder(nodes=[QA, SA])

# Construct the top-level partial order connecting all components
root = StrictPartialOrder(nodes=[SR, CS, po_da_aci, loop_node, choice_node, po_qa_sa, GLA])

# Add the required order edges
root.order.add_edge(SR, CS)
root.order.add_edge(CS, po_da_aci)
root.order.add_edge(po_da_aci, loop_node)
root.order.add_edge(loop_node, choice_node)
root.order.add_edge(choice_node, po_qa_sa)
root.order.add_edge(po_qa_sa, GLA)
