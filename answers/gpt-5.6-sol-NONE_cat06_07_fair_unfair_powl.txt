```python
from pm4py.objects.powl.obj import (
    StrictPartialOrder,
    OperatorPOWL,
    Transition,
)
from pm4py.objects.process_tree.obj import Operator


def build_unfair_hiring_model():
    """
    Potentially unfair model:
    All candidates complete the initial stages, but an XOR selects either
    the standard cultural-fit evaluation or a community-affiliation path
    that may confer an implicit advantage.
    """
    receive_application = Transition(label="ReceiveApplication")
    data_completeness_check = Transition(label="DataCompletenessCheck")
    request_more_info = Transition(label="RequestMoreInfo")

    # Check completeness; if incomplete, request more information and recheck.
    completeness_loop = OperatorPOWL(
        operator=Operator.LOOP,
        children=[data_completeness_check, request_more_info],
    )

    skill_assessment = Transition(label="SkillAssessment")

    cultural_fit_check = Transition(label="CulturalFitCheck")
    community_affiliation_check = Transition(label="CommunityAffiliationCheck")

    # Potential bias point: only one of these evaluation paths is taken.
    cultural_path_xor = OperatorPOWL(
        operator=Operator.XOR,
        children=[
            cultural_fit_check,
            community_affiliation_check,
        ],
    )

    managerial_review = Transition(label="ManagerialReview")
    final_decision = Transition(label="FinalDecision")

    root = StrictPartialOrder(
        nodes=[
            receive_application,
            completeness_loop,
            skill_assessment,
            cultural_path_xor,
            managerial_review,
            final_decision,
        ]
    )

    root.order.add_edge(receive_application, completeness_loop)
    root.order.add_edge(completeness_loop, skill_assessment)
    root.order.add_edge(skill_assessment, cultural_path_xor)
    root.order.add_edge(cultural_path_xor, managerial_review)
    root.order.add_edge(managerial_review, final_decision)

    return root


def build_fair_hiring_model():
    """
    Fairer model:
    Every candidate follows the same cultural-fit evaluation. There is no
    special community-affiliation branch or affiliation-based score uplift.
    """
    receive_application = Transition(label="ReceiveApplication")
    data_completeness_check = Transition(label="DataCompletenessCheck")
    request_more_info = Transition(label="RequestMoreInfo")

    completeness_loop = OperatorPOWL(
        operator=Operator.LOOP,
        children=[data_completeness_check, request_more_info],
    )

    skill_assessment = Transition(label="SkillAssessment")
    cultural_fit_check = Transition(label="CulturalFitCheck")
    managerial_review = Transition(label="ManagerialReview")
    final_decision = Transition(label="FinalDecision")

    root = StrictPartialOrder(
        nodes=[
            receive_application,
            completeness_loop,
            skill_assessment,
            cultural_fit_check,
            managerial_review,
            final_decision,
        ]
    )

    root.order.add_edge(receive_application, completeness_loop)
    root.order.add_edge(completeness_loop, skill_assessment)
    root.order.add_edge(skill_assessment, cultural_fit_check)
    root.order.add_edge(cultural_fit_check, managerial_review)
    root.order.add_edge(managerial_review, final_decision)

    return root


unfair_model = build_unfair_hiring_model()
fair_model = build_fair_hiring_model()
```

Conceptually, the models are:

```text
Potential unfairness:
ReceiveApplication
   *(DataCompletenessCheck, RequestMoreInfo)
   SkillAssessment
   X(CulturalFitCheck, CommunityAffiliationCheck)
   ManagerialReview
   FinalDecision
```

```text
Without unfairness:
ReceiveApplication
   *(DataCompletenessCheck, RequestMoreInfo)
   SkillAssessment
   CulturalFitCheck
   ManagerialReview
   FinalDecision
```

The first model locates the potential bias in the XOR branch: applicants routed through `CommunityAffiliationCheck` may receive an implicit advantage. The second model removes that branch so every applicant undergoes the same `CulturalFitCheck`.
