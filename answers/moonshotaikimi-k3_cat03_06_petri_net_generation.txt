The loops are modeled with XOR-choice places:

- After **Document Verification**, either the claim proceeds to fraud checking or **Re-Document Submission** returns it for another verification.
- After **Fraud Check**, either the claim proceeds to assessment or **In-depth Investigation** returns it for another fraud check.
- The threshold-based approval is represented by an XOR between **Approval** and a silent “approval not required” transition, since an ordinary Petri net cannot directly evaluate the claim amount.

```python
from pm4py.objects.petri_net.obj import PetriNet, Marking
from pm4py.objects.petri_net.utils import petri_utils


def build_insurance_claim_net():
    net = PetriNet(name="insurance_claim_management")

    # ------------------------------------------------------------------
    # Helper functions
    # ------------------------------------------------------------------
    def add_place(name):
        place = PetriNet.Place(name)
        net.places.add(place)
        return place

    def add_transition(name, label=None):
        transition = PetriNet.Transition(name, label)
        net.transitions.add(transition)
        return transition

    def add_arc(source, target):
        petri_utils.add_arc_from_to(source, target, net)

    # ------------------------------------------------------------------
    # Places
    # ------------------------------------------------------------------
    p_start = add_place("start")

    p_claim_filed = add_place("claim_filed")
    p_documents_ready = add_place("documents_ready_for_verification")
    p_documents_checked = add_place("documents_checked")

    p_fraud_check_ready = add_place("ready_for_fraud_check")
    p_fraud_checked = add_place("fraud_checked")

    p_assessment_ready = add_place("ready_for_claim_assessment")
    p_claim_assessed = add_place("claim_assessed")

    p_approval_resolved = add_place("approval_resolved")
    p_paid = add_place("claim_paid")
    p_notified = add_place("customer_notified")

    p_closed = add_place("claim_closed")

    # ------------------------------------------------------------------
    # Activity transitions
    # ------------------------------------------------------------------
    t_c = add_transition("C", "Insurance Claim")
    t_cr = add_transition("CR", "Claim Registration")
    t_dv = add_transition("DV", "Document Verification")
    t_rds = add_transition("RDS", "Re-Document Submission")

    t_fc = add_transition("FC", "Fraud Check")
    t_ii = add_transition("II", "In-depth Investigation")

    t_ca = add_transition("CA", "Claim Assessment")
    t_ap = add_transition("AP", "Approval")
    t_p = add_transition("P", "Payment")
    t_n = add_transition("N", "Notification")
    t_cl = add_transition("CL", "Closure")

    # Silent control-flow transitions.
    # They represent decisions rather than business activities.
    t_documents_accepted = add_transition(
        "tau_documents_accepted", label=None
    )
    t_fraud_check_passed = add_transition(
        "tau_fraud_check_passed", label=None
    )
    t_no_approval_required = add_transition(
        "tau_no_approval_required", label=None
    )

    # ------------------------------------------------------------------
    # Initial sequence: C -> CR -> DV
    # ------------------------------------------------------------------
    add_arc(p_start, t_c)
    add_arc(t_c, p_claim_filed)

    add_arc(p_claim_filed, t_cr)
    add_arc(t_cr, p_documents_ready)

    add_arc(p_documents_ready, t_dv)
    add_arc(t_dv, p_documents_checked)

    # ------------------------------------------------------------------
    # Document-verification decision
    #
    # Accepted documents:
    #   DV -> documents accepted -> FC
    #
    # Incomplete documents:
    #   DV -> RDS -> DV
    # ------------------------------------------------------------------
    add_arc(p_documents_checked, t_documents_accepted)
    add_arc(t_documents_accepted, p_fraud_check_ready)

    add_arc(p_documents_checked, t_rds)
    add_arc(t_rds, p_documents_ready)

    # ------------------------------------------------------------------
    # Fraud-check decision
    #
    # No doubts:
    #   FC -> fraud check passed -> CA
    #
    # Doubts:
    #   FC -> II -> FC
    # ------------------------------------------------------------------
    add_arc(p_fraud_check_ready, t_fc)
    add_arc(t_fc, p_fraud_checked)

    add_arc(p_fraud_checked, t_fraud_check_passed)
    add_arc(t_fraud_check_passed, p_assessment_ready)

    add_arc(p_fraud_checked, t_ii)
    add_arc(t_ii, p_fraud_check_ready)

    # ------------------------------------------------------------------
    # Claim assessment
    # ------------------------------------------------------------------
    add_arc(p_assessment_ready, t_ca)
    add_arc(t_ca, p_claim_assessed)

    # ------------------------------------------------------------------
    # Approval decision
    #
    # Amount above threshold:
    #   CA -> AP -> Payment
    #
    # Amount at or below threshold:
    #   CA -> silent bypass -> Payment
    # ------------------------------------------------------------------
    add_arc(p_claim_assessed, t_ap)
    add_arc(t_ap, p_approval_resolved)

    add_arc(p_claim_assessed, t_no_approval_required)
    add_arc(t_no_approval_required, p_approval_resolved)

    # ------------------------------------------------------------------
    # Final sequence: P -> N -> CL
    # ------------------------------------------------------------------
    add_arc(p_approval_resolved, t_p)
    add_arc(t_p, p_paid)

    add_arc(p_paid, t_n)
    add_arc(t_n, p_notified)

    add_arc(p_notified, t_cl)
    add_arc(t_cl, p_closed)

    # ------------------------------------------------------------------
    # Initial and final markings
    # ------------------------------------------------------------------
    im = Marking()
    im[p_start] = 1

    fm = Marking()
    fm[p_closed] = 1

    return net, im, fm


# The accepting Petri net is represented by the tuple (net, im, fm).
net, im, fm = build_insurance_claim_net()
```
