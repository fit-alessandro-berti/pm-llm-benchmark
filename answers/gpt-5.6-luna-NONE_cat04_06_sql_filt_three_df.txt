```sql
WITH ordered_events AS (
    SELECT
        case_id,
        activity,
        "timestamp",
        LEAD(activity, 1) OVER (
            PARTITION BY case_id
            ORDER BY "timestamp"
        ) AS next_activity,
        LEAD(activity, 2) OVER (
            PARTITION BY case_id
            ORDER BY "timestamp"
        ) AS activity_after_next,
        LEAD("timestamp", 1) OVER (
            PARTITION BY case_id
            ORDER BY "timestamp"
        ) AS next_timestamp,
        LEAD("timestamp", 2) OVER (
            PARTITION BY case_id
            ORDER BY "timestamp"
        ) AS confirm_timestamp
    FROM event_log
),
cases_to_exclude AS (
    SELECT DISTINCT case_id
    FROM ordered_events
    WHERE activity = 'Approve Purchase Requisition'
      AND next_activity = 'Create Order'
      AND activity_after_next = 'Confirm Order'
      AND confirm_timestamp - next_timestamp > INTERVAL '5 days'
)
SELECT e.*
FROM event_log AS e
WHERE NOT EXISTS (
    SELECT 1
    FROM cases_to_exclude AS x
    WHERE x.case_id = e.case_id
);
```

The `LEAD` functions identify three directly consecutive events within each case. The `cases_to_exclude` CTE selects cases where the interval between `Create Order` and `Confirm Order` exceeds five days, and the final query removes all events belonging to those cases.
