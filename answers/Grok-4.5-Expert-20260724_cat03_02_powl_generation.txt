from pm4py.objects.powl.obj import StrictPartialOrder, OperatorPOWL, Transition, SilentTransition
from pm4py.objects.process_tree.obj import Operator

# Atomic activities
SR = Transition(label="SR")
CS = Transition(label="CS")
DA = Transition(label="DA")
ACI = Transition(label="ACI")
BNC = Transition(label="BNC")
DR = Transition(label="D&R")
CD = Transition(label="CD")
MD = Transition(label="MD")
QA = Transition(label="QA")
SA = Transition(label="SA")
GLA = Transition(label="GLA")

# Loop: execute BNC, then either exit or perform D&R and repeat BNC
loop = OperatorPOWL(operator=Operator.LOOP, children=[BNC, DR])

# Exclusive choice between the two deployment methods
xor = OperatorPOWL(operator=Operator.XOR, children=[CD, MD])

# Final tasks: QA and SA concurrent, both must precede GLA
post = StrictPartialOrder(nodes=[QA, SA, GLA])
post.order.add_edge(QA, GLA)
post.order.add_edge(SA, GLA)

# Deployment choice followed by the post-deployment partial order
after_deploy = StrictPartialOrder(nodes=[xor, post])
after_deploy.order.add_edge(xor, post)

# Configuration loop followed by deployment + post tasks
after_config = StrictPartialOrder(nodes=[loop, after_deploy])
after_config.order.add_edge(loop, after_deploy)

# Initial parallel tasks (DA and ACI concurrent)
init = StrictPartialOrder(nodes=[DA, ACI])

# After CS: parallel init, then the configuration/deployment chain
after_cs = StrictPartialOrder(nodes=[init, after_config])
after_cs.order.add_edge(init, after_config)

# Top-level partial order: SR → CS → rest of the process
root = StrictPartialOrder(nodes=[SR, CS, after_cs])
root.order.add_edge(SR, CS)
root.order.add_edge(CS, after_cs)
