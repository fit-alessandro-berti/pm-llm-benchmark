**Query 1: Identify cases where goods were shipped before Credit Check or Stock Validation**
This query checks if the timestamp for "Ship Goods" is earlier than "Perform Credit Check" or "Validate Stock".
```sql
WITH EventTimes AS (
    SELECT 
        case_id,
        MAX(CASE WHEN activity = 'Ship Goods' THEN timestamp END) AS ship_time,
        MAX(CASE WHEN activity = 'Perform Credit Check' THEN timestamp END) AS credit_time,
        MAX(CASE WHEN activity = 'Validate Stock' THEN timestamp END) AS stock_time
    FROM order_event_log
    GROUP BY case_id
)
SELECT 
    e.case_id,
    o.order_type,
    o.order_value,
    e.ship_time,
    e.credit_time,
    e.stock_time
FROM EventTimes e
JOIN orders o ON e.case_id = o.case_id
WHERE (e.ship_time < e.credit_time) OR (e.ship_time < e.stock_time)
ORDER BY e.case_id;
```

**Query 2: Identify cases where Payment was received before the Invoice was issued**
```sql
WITH FinanceTimes AS (
    SELECT 
        case_id,
        MAX(CASE WHEN activity = 'Issue Invoice' THEN timestamp END) AS invoice_time,
        MAX(CASE WHEN activity = 'Receive Payment' THEN timestamp END) AS payment_time
    FROM order_event_log
    GROUP BY case_id
)
SELECT 
    f.case_id,
    f.invoice_time,
    f.payment_time
FROM FinanceTimes f
WHERE f.payment_time < f.invoice_time
ORDER BY f.case_id;
```

**Query 3: Find cases missing mandatory activities (Credit Check or Stock Validation)**
```sql
SELECT 
    o.case_id,
    o.order_type,
    o.order_value
FROM orders o
WHERE NOT EXISTS (
    SELECT 1 FROM order_event_log e 
    WHERE e.case_id = o.case_id 
    AND e.activity = 'Perform Credit Check'
)
OR NOT EXISTS (
    SELECT 1 FROM order_event_log e 
    WHERE e.case_id = o.case_id 
    AND e.activity = 'Validate Stock'
)
ORDER BY o.case_id;
```

**Query 4: Correlate out-of-sequence events with order types (e.g., are priority orders more anomalous?)**
This query flags any case where activities do not follow the expected strict chronological order.
```sql
WITH RankedEvents AS (
    SELECT 
        case_id,
        activity,
        timestamp,
        ROW_NUMBER() OVER(PARTITION BY case_id ORDER BY timestamp) as time_rank,
        CASE activity
            WHEN 'Register Order' THEN 1
            WHEN 'Perform Credit Check' THEN 2
            WHEN 'Validate Stock' THEN 3
            WHEN 'Confirm Shipment' THEN 4
            WHEN 'Ship Goods' THEN 5
            WHEN 'Issue Invoice' THEN 6
            WHEN 'Receive Payment' THEN 7
        END as expected_rank
    FROM order_event_log
)
SELECT 
    r.case_id,
    o.order_type,
    o.order_value,
    COUNT(*) as out_of_order_events
FROM RankedEvents r
JOIN orders o ON r.case_id = o.case_id
WHERE r.time_rank != r.expected_rank
GROUP BY r.case_id, o.order_type, o.order_value
ORDER BY out_of_order_events DESC, r.case_id;
```

**Query 5: Investigate resources/departments involved in anomalous "Ship Goods" events**
This query joins with the `resources` table to see if a specific department or role is responsible for shipping goods before checks are completed.
```sql
SELECT 
    e.case_id,
    e.timestamp AS ship_time,
    e.resource,
    r.role,
    r.department,
    e.additional_info
FROM order_event_log e
JOIN resources r ON e.resource = r.resource_id
WHERE e.activity = 'Ship Goods'
AND e.case_id IN (
    SELECT case_id 
    FROM order_event_log 
    WHERE activity = 'Perform Credit Check' 
    AND timestamp > e.timestamp
)
ORDER BY r.department, r.role;
```
