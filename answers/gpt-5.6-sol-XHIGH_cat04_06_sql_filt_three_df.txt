```sql
WITH sequenced AS (
    SELECT
        e.*,
        LEAD(activity, 1) OVER event_order AS next_activity,
        LEAD(activity, 2) OVER event_order AS second_next_activity,
        LEAD("timestamp", 1) OVER event_order AS create_order_time,
        LEAD("timestamp", 2) OVER event_order AS confirm_order_time
    FROM event_log AS e
    WINDOW event_order AS (
        PARTITION BY case_id
        ORDER BY "timestamp"
    )
),
excluded_cases AS (
    SELECT DISTINCT case_id
    FROM sequenced
    WHERE activity = 'Approve Purchase Requisition'
      AND next_activity = 'Create Order'
      AND second_next_activity = 'Confirm Order'
      AND confirm_order_time - create_order_time > INTERVAL '5 days'
)
SELECT e.*
FROM event_log AS e
WHERE NOT EXISTS (
    SELECT 1
    FROM excluded_cases AS x
    WHERE x.case_id IS NOT DISTINCT FROM e.case_id
)
ORDER BY e.case_id, e."timestamp";
```

`LEAD` checks the next two events within each case, so the activities must be directly successive. If multiple events can have the same timestamp, add a deterministic event-order column, such as `event_id`, to both `ORDER BY` clauses (for example, `ORDER BY "timestamp", event_id`).
