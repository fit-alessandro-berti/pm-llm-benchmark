```python
import pm4py
from pm4py.objects.powl.obj import StrictPartialOrder, OperatorPOWL, Transition, SilentTransition
from pm4py.objects.process_tree.obj import Operator

# Shared activity definitions
receive = Transition(label="ReceiveApplication")
data_check = Transition(label="DataCompletenessCheck")
request_info = Transition(label="RequestMoreInfo")
skill = Transition(label="SkillAssessment")
cultural = Transition(label="CulturalFitCheck")
community = Transition(label="CommunityAffiliationCheck")
managerial = Transition(label="ManagerialReview")
final = Transition(label="FinalDecision")

# ============================================================
# 1. POWL Model WITH potential unfairness
#    (XOR after skill assessment: standard cultural fit
#     OR community-affiliation path that introduces bias)
# ============================================================

# Loop: execute DataCompletenessCheck; either exit or
# execute RequestMoreInfo and repeat the check
loop = OperatorPOWL(operator=Operator.LOOP, children=[data_check, request_info])

# XOR choice that can introduce bias
xor_biased = OperatorPOWL(operator=Operator.XOR, children=[cultural, community])

root_unfair = StrictPartialOrder(nodes=[receive, loop, skill, xor_biased, managerial, final])
root_unfair.order.add_edge(receive, loop)
root_unfair.order.add_edge(loop, skill)
root_unfair.order.add_edge(skill, xor_biased)
root_unfair.order.add_edge(xor_biased, managerial)
root_unfair.order.add_edge(managerial, final)

# ============================================================
# 2. POWL Model WITHOUT unfairness
#    (identical structure except every applicant follows
#     the same CulturalFitCheck – no preferential branch)
# ============================================================

# Same data-completeness loop
loop_fair = OperatorPOWL(operator=Operator.LOOP, children=[data_check, request_info])

root_fair = StrictPartialOrder(nodes=[receive, loop_fair, skill, cultural, managerial, final])
root_fair.order.add_edge(receive, loop_fair)
root_fair.order.add_edge(loop_fair, skill)
root_fair.order.add_edge(skill, cultural)
root_fair.order.add_edge(cultural, managerial)
root_fair.order.add_edge(managerial, final)
```
