### F. Detect Cases Where Financial Events Are Out of Order or Missing

```sql
WITH case_timeline AS (
  SELECT case_id,
         MIN(CASE WHEN activity = 'Issue Invoice' THEN timestamp END) AS invoice_ts,
         MIN(CASE WHEN activity = 'Receive Payment' THEN timestamp END) AS payment_ts,
         MIN(CASE WHEN activity = 'Register Order' THEN timestamp END) AS reg_ts
  FROM order_event_log
  GROUP BY case_id
)
SELECT case_id,
       reg_ts,
       invoice_ts,
       payment_ts,
       CASE WHEN invoice_ts IS NULL THEN 'MISSING_INVOICE'
            WHEN payment_ts < invoice_ts THEN 'PAYMENT_BEFORE_INVOICE'
            ELSE 'NORMAL' END AS finance_anomaly
FROM case_timeline;
```

### G. Check If Specific Resources Repeat in Violation Patterns
```sql
SELECT r.resource_id, r.role, r.department, COUNT(DISTINCT e.case_id) AS anomalous_cases
FROM order_event_log e
JOIN resources r ON e.resource = r.resource_id
WHERE e.additional_info ILIKE '%attempted_early=%'
   OR e.additional_info ILIKE '%late_confirmation=%'
   OR e.additional_info ILIKE '%scheduled=N%'
GROUP BY r.resource_id, r.role, r.department
ORDER BY anomalous_cases DESC;
```
