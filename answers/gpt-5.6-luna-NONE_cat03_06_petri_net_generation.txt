```python
from pm4py.objects.petri_net.obj import PetriNet, Marking
from pm4py.objects.petri_net.utils import petri_utils

# Create the Petri net
net = PetriNet("Insurance Claim Management")

# ---------------------------------------------------------------------
# Places: states between activities
# ---------------------------------------------------------------------
p_start = PetriNet.Place("claim_received")
p_after_c = PetriNet.Place("after_claim")
p_after_cr = PetriNet.Place("after_registration")
p_after_dv = PetriNet.Place("after_document_verification")
p_after_rds = PetriNet.Place("after_resubmission")
p_after_fc = PetriNet.Place("after_fraud_check")
p_after_ii = PetriNet.Place("after_investigation")
p_after_ca = PetriNet.Place("after_assessment")
p_after_ap = PetriNet.Place("after_approval")
p_after_p = PetriNet.Place("after_payment")
p_after_n = PetriNet.Place("after_notification")
p_closed = PetriNet.Place("claim_closed")

for place in [
    p_start,
    p_after_c,
    p_after_cr,
    p_after_dv,
    p_after_rds,
    p_after_fc,
    p_after_ii,
    p_after_ca,
    p_after_ap,
    p_after_p,
    p_after_n,
    p_closed,
]:
    net.places.add(place)

# ---------------------------------------------------------------------
# Transitions: activities
# ---------------------------------------------------------------------
t_c = PetriNet.Transition("C", "Insurance Claim")
t_cr = PetriNet.Transition("CR", "Claim Registration")
t_dv = PetriNet.Transition("DV", "Document Verification")
t_rds = PetriNet.Transition("RDS", "Re-Document Submission")
t_fc = PetriNet.Transition("FC", "Fraud Check")
t_ii = PetriNet.Transition("II", "In-depth Investigation")
t_ca = PetriNet.Transition("CA", "Claim Assessment")
t_ap = PetriNet.Transition("AP", "Approval")
t_p = PetriNet.Transition("P", "Payment")
t_n = PetriNet.Transition("N", "Notification")
t_cl = PetriNet.Transition("CL", "Closure")

for transition in [
    t_c,
    t_cr,
    t_dv,
    t_rds,
    t_fc,
    t_ii,
    t_ca,
    t_ap,
    t_p,
    t_n,
    t_cl,
]:
    net.transitions.add(transition)

# Helper function for adding arcs
def connect(source, target):
    petri_utils.add_arc_from_to(source, target, net)

# ---------------------------------------------------------------------
# Main process sequence
# ---------------------------------------------------------------------
connect(p_start, t_c)
connect(t_c, p_after_c)

connect(p_after_c, t_cr)
connect(t_cr, p_after_cr)

connect(p_after_cr, t_dv)
connect(t_dv, p_after_dv)

connect(p_after_dv, t_fc)
connect(t_fc, p_after_fc)

connect(p_after_fc, t_ca)
connect(t_ca, p_after_ca)

connect(p_after_ca, t_ap)
connect(t_ap, p_after_ap)

connect(p_after_ap, t_p)
connect(t_p, p_after_p)

connect(p_after_p, t_n)
connect(t_n, p_after_n)

connect(p_after_n, t_cl)
connect(t_cl, p_closed)

# ---------------------------------------------------------------------
# Document re-submission loop
#
# After DV, RDS may be executed. After resubmission, the process returns
# to DV, allowing the loop to occur multiple times.
# ---------------------------------------------------------------------
connect(p_after_dv, t_rds)
connect(t_rds, p_after_rds)
connect(p_after_rds, t_dv)

# ---------------------------------------------------------------------
# Fraud investigation loop
#
# After FC, an investigation may be required. After II, the process
# returns to FC, allowing repeated investigations.
# ---------------------------------------------------------------------
connect(p_after_fc, t_ii)
connect(t_ii, p_after_ii)
connect(p_after_ii, t_fc)

# ---------------------------------------------------------------------
# Initial and final markings
# ---------------------------------------------------------------------
im = Marking()
im[p_start] = 1

fm = Marking()
fm[p_closed] = 1
```

The net starts with one token in `claim_received`. A token reaches `claim_closed` only after the sequence:

```text
C  CR  DV  FC  CA  AP  P  N  CL
```

The two loops allow repeated processing:

```text
DV  RDS  DV
FC  II  FC
```
