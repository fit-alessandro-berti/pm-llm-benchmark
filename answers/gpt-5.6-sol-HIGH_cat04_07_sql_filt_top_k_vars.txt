```sql
WITH case_sequences AS (
    -- Build the ordered activity sequence for each case
    SELECT
        case_id,
        list(activity ORDER BY "timestamp") AS activity_sequence
    FROM event_log
    GROUP BY case_id
),

variant_counts AS (
    -- Count how many cases follow each sequence
    SELECT
        activity_sequence,
        COUNT(*) AS case_count
    FROM case_sequences
    GROUP BY activity_sequence
),

top_variants AS (
    -- Select the K most frequent variants
    SELECT
        activity_sequence,
        case_count
    FROM variant_counts
    ORDER BY
        case_count DESC,
        activity_sequence   -- deterministic tie-breaker
    LIMIT $k
),

top_variant_cases AS (
    -- Find all cases belonging to the selected variants
    SELECT cs.case_id
    FROM case_sequences AS cs
    INNER JOIN top_variants AS tv
        ON cs.activity_sequence = tv.activity_sequence
)

-- Return the original events for those cases
SELECT e.*
FROM event_log AS e
INNER JOIN top_variant_cases AS tc
    ON e.case_id = tc.case_id
ORDER BY
    e.case_id,
    e."timestamp";
```

`$k` is the DuckDB parameter for the desired number of variants and can be replaced with a literal such as `LIMIT 10`.

If multiple events in the same case can have identical timestamps, add a stable event identifier as a secondary ordering expression, for example:

```sql
list(activity ORDER BY "timestamp", event_id)
```
