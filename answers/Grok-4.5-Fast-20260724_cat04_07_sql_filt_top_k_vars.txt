```sql
-- Parameter: replace :K with the desired integer (e.g. 5)
WITH case_variants AS (
    SELECT
        case_id,
        list(activity ORDER BY timestamp) AS variant
    FROM event_log
    GROUP BY case_id
),
top_variants AS (
    SELECT variant
    FROM case_variants
    GROUP BY variant
    ORDER BY count(*) DESC
    LIMIT :K
)
SELECT el.*
FROM event_log el
JOIN case_variants cv
  ON el.case_id = cv.case_id
WHERE cv.variant IN (SELECT variant FROM top_variants)
ORDER BY el.case_id, el.timestamp;
```
