The net below uses silent transitions (`label=None`) for routing decisions that are not recorded activities, such as continuing after successful document verification or bypassing approval for claims below the threshold.

```python
from pm4py.objects.petri_net.obj import PetriNet, Marking
from pm4py.objects.petri_net.utils import petri_utils


# ------------------------------------------------------------------
# Create the Petri net
# ------------------------------------------------------------------
net = PetriNet(name="insurance_claim_management")


# ------------------------------------------------------------------
# Places
# ------------------------------------------------------------------
p_start = PetriNet.Place("p_start")
p_claim_filed = PetriNet.Place("p_claim_filed")
p_registered = PetriNet.Place("p_registered")
p_documents_checked = PetriNet.Place("p_documents_checked")
p_ready_for_fraud_check = PetriNet.Place("p_ready_for_fraud_check")
p_fraud_checked = PetriNet.Place("p_fraud_checked")
p_assessed = PetriNet.Place("p_assessed")
p_ready_for_payment = PetriNet.Place("p_ready_for_payment")
p_paid = PetriNet.Place("p_paid")
p_notified = PetriNet.Place("p_notified")
p_closed = PetriNet.Place("p_closed")

places = [
    p_start,
    p_claim_filed,
    p_registered,
    p_documents_checked,
    p_ready_for_fraud_check,
    p_fraud_checked,
    p_assessed,
    p_ready_for_payment,
    p_paid,
    p_notified,
    p_closed,
]

for place in places:
    net.places.add(place)


# ------------------------------------------------------------------
# Visible activity transitions
# ------------------------------------------------------------------
t_claim = PetriNet.Transition(
    "t_claim",
    "Insurance Claim (C)"
)
t_registration = PetriNet.Transition(
    "t_registration",
    "Claim Registration (CR)"
)
t_document_verification = PetriNet.Transition(
    "t_document_verification",
    "Document Verification (DV)"
)
t_document_resubmission = PetriNet.Transition(
    "t_document_resubmission",
    "Re-Document Submission (RDS)"
)
t_fraud_check = PetriNet.Transition(
    "t_fraud_check",
    "Fraud Check (FC)"
)
t_in_depth_investigation = PetriNet.Transition(
    "t_in_depth_investigation",
    "In-depth Investigation (II)"
)
t_claim_assessment = PetriNet.Transition(
    "t_claim_assessment",
    "Claim Assessment (CA)"
)
t_approval = PetriNet.Transition(
    "t_approval",
    "Approval (AP)"
)
t_payment = PetriNet.Transition(
    "t_payment",
    "Payment (P)"
)
t_notification = PetriNet.Transition(
    "t_notification",
    "Notification (N)"
)
t_closure = PetriNet.Transition(
    "t_closure",
    "Closure (CL)"
)


# ------------------------------------------------------------------
# Silent routing transitions
# ------------------------------------------------------------------

# Taken when the documents are complete after verification.
t_documents_complete = PetriNet.Transition(
    "t_documents_complete",
    None
)

# Taken when approval is unnecessary because the claim amount is
# below the approval threshold.
t_skip_approval = PetriNet.Transition(
    "t_skip_approval",
    None
)

transitions = [
    t_claim,
    t_registration,
    t_document_verification,
    t_document_resubmission,
    t_documents_complete,
    t_fraud_check,
    t_in_depth_investigation,
    t_claim_assessment,
    t_approval,
    t_skip_approval,
    t_payment,
    t_notification,
    t_closure,
]

for transition in transitions:
    net.transitions.add(transition)


# ------------------------------------------------------------------
# Arcs: claim filing and registration
# ------------------------------------------------------------------
petri_utils.add_arc_from_to(p_start, t_claim, net)
petri_utils.add_arc_from_to(t_claim, p_claim_filed, net)

petri_utils.add_arc_from_to(p_claim_filed, t_registration, net)
petri_utils.add_arc_from_to(t_registration, p_registered, net)


# ------------------------------------------------------------------
# Document verification and re-submission loop
#
# DV -> documents checked
# If incomplete: RDS -> DV can be performed again
# If complete: continue silently to fraud checking
# ------------------------------------------------------------------
petri_utils.add_arc_from_to(p_registered, t_document_verification, net)
petri_utils.add_arc_from_to(
    t_document_verification, p_documents_checked, net
)

petri_utils.add_arc_from_to(
    p_documents_checked, t_document_resubmission, net
)
petri_utils.add_arc_from_to(
    t_document_resubmission, p_registered, net
)

petri_utils.add_arc_from_to(
    p_documents_checked, t_documents_complete, net
)
petri_utils.add_arc_from_to(
    t_documents_complete, p_ready_for_fraud_check, net
)


# ------------------------------------------------------------------
# Fraud check and in-depth investigation loop
#
# FC -> fraud checked
# If doubts remain: II -> FC can be performed again
# Otherwise: continue with claim assessment
# ------------------------------------------------------------------
petri_utils.add_arc_from_to(
    p_ready_for_fraud_check, t_fraud_check, net
)
petri_utils.add_arc_from_to(
    t_fraud_check, p_fraud_checked, net
)

petri_utils.add_arc_from_to(
    p_fraud_checked, t_in_depth_investigation, net
)
petri_utils.add_arc_from_to(
    t_in_depth_investigation, p_ready_for_fraud_check, net
)

petri_utils.add_arc_from_to(
    p_fraud_checked, t_claim_assessment, net
)
petri_utils.add_arc_from_to(
    t_claim_assessment, p_assessed, net
)


# ------------------------------------------------------------------
# Conditional approval
#
# High-value claim: AP is executed.
# Low-value claim: silent transition skips AP.
# ------------------------------------------------------------------
petri_utils.add_arc_from_to(p_assessed, t_approval, net)
petri_utils.add_arc_from_to(
    t_approval, p_ready_for_payment, net
)

petri_utils.add_arc_from_to(
    p_assessed, t_skip_approval, net
)
petri_utils.add_arc_from_to(
    t_skip_approval, p_ready_for_payment, net
)


# ------------------------------------------------------------------
# Payment, notification, and closure
# ------------------------------------------------------------------
petri_utils.add_arc_from_to(
    p_ready_for_payment, t_payment, net
)
petri_utils.add_arc_from_to(t_payment, p_paid, net)

petri_utils.add_arc_from_to(p_paid, t_notification, net)
petri_utils.add_arc_from_to(t_notification, p_notified, net)

petri_utils.add_arc_from_to(p_notified, t_closure, net)
petri_utils.add_arc_from_to(t_closure, p_closed, net)


# ------------------------------------------------------------------
# Initial and final markings
# ------------------------------------------------------------------
im = Marking()
im[p_start] = 1

fm = Marking()
fm[p_closed] = 1
```

The document loop permits traces such as:

```text
DV, RDS, DV, RDS, DV, FC, ...
```

The investigation loop permits:

```text
FC, II, FC, II, FC, CA, ...
```

Because ordinary Petri nets do not directly evaluate claim amounts or document/fraud data, these conditions are represented as nondeterministic routing choices.
