```sql
WITH ordered_events AS (
    SELECT
        e.*,
        LEAD(activity, 1) OVER (
            PARTITION BY case_id
            ORDER BY "timestamp"
        ) AS next_activity,
        LEAD(activity, 2) OVER (
            PARTITION BY case_id
            ORDER BY "timestamp"
        ) AS activity_after_next,
        LEAD("timestamp", 1) OVER (
            PARTITION BY case_id
            ORDER BY "timestamp"
        ) AS create_order_timestamp,
        LEAD("timestamp", 2) OVER (
            PARTITION BY case_id
            ORDER BY "timestamp"
        ) AS confirm_order_timestamp
    FROM event_log AS e
),
cases_to_exclude AS (
    SELECT DISTINCT case_id
    FROM ordered_events
    WHERE activity = 'Approve Purchase Requisition'
      AND next_activity = 'Create Order'
      AND activity_after_next = 'Confirm Order'
      AND confirm_order_timestamp - create_order_timestamp
            > INTERVAL '5 days'
)
SELECT e.*
FROM event_log AS e
WHERE NOT EXISTS (
    SELECT 1
    FROM cases_to_exclude AS c
    WHERE c.case_id IS NOT DISTINCT FROM e.case_id
);
```

The two `LEAD` expressions ensure that the three activities are adjacent events within the same case. The final `NOT EXISTS` removes every event belonging to any case that satisfies both the sequence and timing conditions.

If the table has a unique event-order column, such as `event_id`, use:

```sql
ORDER BY "timestamp", event_id
```

in each window function to make ordering deterministic when multiple events have the same timestamp.
