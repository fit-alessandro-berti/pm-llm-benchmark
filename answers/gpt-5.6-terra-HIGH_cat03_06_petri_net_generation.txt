```python
from pm4py.objects.petri_net.obj import PetriNet, Marking
from pm4py.objects.petri_net.utils import petri_utils

# Create the Petri net
net = PetriNet(name="Insurance Claim Management Process")

# -------------------------------------------------------------------
# Places: represent states between activities
# -------------------------------------------------------------------
p_start = PetriNet.Place("start")
p_claim_filed = PetriNet.Place("claim_filed")
p_registered = PetriNet.Place("registered")
p_waiting_for_documents = PetriNet.Place("waiting_for_documents")
p_documents_verified = PetriNet.Place("documents_verified")
p_fraud_checked = PetriNet.Place("fraud_checked")
p_assessed = PetriNet.Place("assessed")
p_approved = PetriNet.Place("approved")
p_paid = PetriNet.Place("paid")
p_notified = PetriNet.Place("notified")
p_closed = PetriNet.Place("closed")

net.places.update({
    p_start,
    p_claim_filed,
    p_registered,
    p_waiting_for_documents,
    p_documents_verified,
    p_fraud_checked,
    p_assessed,
    p_approved,
    p_paid,
    p_notified,
    p_closed
})

# -------------------------------------------------------------------
# Transitions: represent process activities
# -------------------------------------------------------------------
t_claim = PetriNet.Transition("C", "Insurance Claim (C)")
t_claim_registration = PetriNet.Transition("CR", "Claim Registration (CR)")
t_document_verification = PetriNet.Transition("DV", "Document Verification (DV)")
t_re_document_submission = PetriNet.Transition(
    "RDS", "Re-Document Submission (RDS)"
)
t_fraud_check = PetriNet.Transition("FC", "Fraud Check (FC)")
t_in_depth_investigation = PetriNet.Transition(
    "II", "In-depth Investigation (II)"
)
t_claim_assessment = PetriNet.Transition("CA", "Claim Assessment (CA)")
t_approval = PetriNet.Transition("AP", "Approval (AP)")
t_payment = PetriNet.Transition("P", "Payment (P)")
t_notification = PetriNet.Transition("N", "Notification (N)")
t_closure = PetriNet.Transition("CL", "Closure (CL)")

net.transitions.update({
    t_claim,
    t_claim_registration,
    t_document_verification,
    t_re_document_submission,
    t_fraud_check,
    t_in_depth_investigation,
    t_claim_assessment,
    t_approval,
    t_payment,
    t_notification,
    t_closure
})

# -------------------------------------------------------------------
# Main process flow
# -------------------------------------------------------------------

# New insurance claim arrives
petri_utils.add_arc_from_to(p_start, t_claim, net)
petri_utils.add_arc_from_to(t_claim, p_claim_filed, net)

# Claim registration
petri_utils.add_arc_from_to(p_claim_filed, t_claim_registration, net)
petri_utils.add_arc_from_to(t_claim_registration, p_registered, net)

# Document verification
petri_utils.add_arc_from_to(p_registered, t_document_verification, net)
petri_utils.add_arc_from_to(t_document_verification, p_documents_verified, net)

# If documents are incomplete, they can be re-submitted.
# The token returns to the state before document verification.
petri_utils.add_arc_from_to(p_documents_verified, t_re_document_submission, net)
petri_utils.add_arc_from_to(t_re_document_submission, p_registered, net)

# Fraud check
petri_utils.add_arc_from_to(p_documents_verified, t_fraud_check, net)
petri_utils.add_arc_from_to(t_fraud_check, p_fraud_checked, net)

# If fraud doubts remain, an in-depth investigation can occur.
# It loops back to the fraud-check stage and may occur repeatedly.
petri_utils.add_arc_from_to(p_fraud_checked, t_in_depth_investigation, net)
petri_utils.add_arc_from_to(t_in_depth_investigation, p_documents_verified, net)

# Claim assessment after fraud check/investigation is completed
petri_utils.add_arc_from_to(p_fraud_checked, t_claim_assessment, net)
petri_utils.add_arc_from_to(t_claim_assessment, p_assessed, net)

# Approval
petri_utils.add_arc_from_to(p_assessed, t_approval, net)
petri_utils.add_arc_from_to(t_approval, p_approved, net)

# Payment
petri_utils.add_arc_from_to(p_approved, t_payment, net)
petri_utils.add_arc_from_to(t_payment, p_paid, net)

# Customer notification
petri_utils.add_arc_from_to(p_paid, t_notification, net)
petri_utils.add_arc_from_to(t_notification, p_notified, net)

# Claim closure
petri_utils.add_arc_from_to(p_notified, t_closure, net)
petri_utils.add_arc_from_to(t_closure, p_closed, net)

# -------------------------------------------------------------------
# Initial and final markings
# -------------------------------------------------------------------

# A new claim is available at the start of the process
im = Marking()
im[p_start] = 1

# The process is complete when the claim is closed
fm = Marking()
fm[p_closed] = 1
```

The `RDS` transition creates a loop back to the document-verification stage, while the `II` transition allows repeated investigation and fraud checking before the claim can proceed to assessment.
