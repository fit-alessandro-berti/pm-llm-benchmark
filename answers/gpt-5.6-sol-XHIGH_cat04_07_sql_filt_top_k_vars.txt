```sql
WITH case_sequences AS (
    -- One ordered activity sequence per case
    SELECT
        case_id,
        list(activity ORDER BY "timestamp") AS activity_sequence
    FROM event_log
    GROUP BY case_id
),
variant_frequencies AS (
    -- Count the cases following each sequence
    SELECT
        activity_sequence,
        COUNT(*) AS case_count
    FROM case_sequences
    GROUP BY activity_sequence
),
top_k_variants AS (
    -- Select exactly the K most frequent variants
    SELECT
        activity_sequence,
        case_count
    FROM variant_frequencies
    ORDER BY case_count DESC, activity_sequence
    LIMIT $k
),
top_variant_cases AS (
    -- Identify the cases belonging to those variants
    SELECT cs.case_id
    FROM case_sequences AS cs
    JOIN top_k_variants AS tv
      ON cs.activity_sequence = tv.activity_sequence
)
-- Return the original events for the selected cases
SELECT el.*
FROM event_log AS el
JOIN top_variant_cases AS tc
  ON el.case_id = tc.case_id
ORDER BY el.case_id, el."timestamp";
```

`$k` is a bind parameter; it can also be replaced with a literal such as `LIMIT 10`.

The secondary ordering by `activity_sequence` makes selection deterministic when multiple variants have the same frequency. If events within a case can have identical timestamps, an additional stable column such as `event_id` should be added to the aggregate ordering:

```sql
list(activity ORDER BY "timestamp", event_id)
```
