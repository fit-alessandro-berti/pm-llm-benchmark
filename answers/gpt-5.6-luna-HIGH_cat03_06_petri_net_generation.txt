```python
from pm4py.objects.petri_net.obj import PetriNet, Marking
from pm4py.objects.petri_net.utils import petri_utils


# ---------------------------------------------------------------------
# Create the Petri net
# ---------------------------------------------------------------------
net = PetriNet(name="insurance_claim_management")

# Places: states between activities
p_start = PetriNet.Place("new_claim")
p_after_c = PetriNet.Place("claim_filed")
p_after_cr = PetriNet.Place("claim_registered")
p_after_dv = PetriNet.Place("documents_verified")
p_after_fc = PetriNet.Place("fraud_check_completed")
p_after_ca = PetriNet.Place("claim_assessed")
p_after_ap = PetriNet.Place("claim_approved")
p_after_p = PetriNet.Place("claim_paid")
p_after_n = PetriNet.Place("customer_notified")
p_closed = PetriNet.Place("claim_closed")

for place in [
    p_start,
    p_after_c,
    p_after_cr,
    p_after_dv,
    p_after_fc,
    p_after_ca,
    p_after_ap,
    p_after_p,
    p_after_n,
    p_closed,
]:
    net.places.add(place)


# Helper for adding a transition and its input/output arcs
def add_transition(name, label, source_place, target_place):
    transition = PetriNet.Transition(name=name, label=label)
    net.transitions.add(transition)

    petri_utils.add_arc_from_to(source_place, transition, net)
    petri_utils.add_arc_from_to(transition, target_place, net)

    return transition


# ---------------------------------------------------------------------
# Main process
# ---------------------------------------------------------------------

# C: Insurance Claim
add_transition(
    "C",
    "Insurance Claim (C)",
    p_start,
    p_after_c,
)

# CR: Claim Registration
add_transition(
    "CR",
    "Claim Registration (CR)",
    p_after_c,
    p_after_cr,
)

# DV: Document Verification
add_transition(
    "DV",
    "Document Verification (DV)",
    p_after_cr,
    p_after_dv,
)

# FC: Fraud Check
# FC and RDS are alternative transitions enabled after DV.
add_transition(
    "FC",
    "Fraud Check (FC)",
    p_after_dv,
    p_after_fc,
)

# RDS: Re-Document Submission
# After resubmission, the claim returns to the document-verification state.
add_transition(
    "RDS",
    "Re-Document Submission (RDS)",
    p_after_dv,
    p_after_cr,
)

# II: In-depth Investigation
# II is a self-loop, so it may occur multiple times before CA.
t_ii = PetriNet.Transition(
    name="II",
    label="In-depth Investigation (II)",
)
net.transitions.add(t_ii)

petri_utils.add_arc_from_to(p_after_fc, t_ii, net)
petri_utils.add_arc_from_to(t_ii, p_after_fc, net)

# CA: Claim Assessment
# CA and II are alternative transitions after the fraud check.
add_transition(
    "CA",
    "Claim Assessment (CA)",
    p_after_fc,
    p_after_ca,
)

# AP: Approval
add_transition(
    "AP",
    "Approval (AP)",
    p_after_ca,
    p_after_ap,
)

# Optional approval bypass:
# In a plain Petri net, the threshold condition is represented abstractly
# by this invisible transition. A data-aware implementation would enable
# this transition only when approval is not required.
t_skip_ap = PetriNet.Transition(
    name="SKIP_AP",
    label=None,  # invisible transition
)
net.transitions.add(t_skip_ap)
petri_utils.add_arc_from_to(p_after_ca, t_skip_ap, net)
petri_utils.add_arc_from_to(t_skip_ap, p_after_ap, net)

# P: Payment
add_transition(
    "P",
    "Payment (P)",
    p_after_ap,
    p_after_p,
)

# N: Notification
add_transition(
    "N",
    "Notification (N)",
    p_after_p,
    p_after_n,
)

# CL: Closure
add_transition(
    "CL",
    "Closure (CL)",
    p_after_n,
    p_closed,
)


# ---------------------------------------------------------------------
# Initial and final markings
# ---------------------------------------------------------------------

im = Marking()
im[p_start] = 1

fm = Marking()
fm[p_closed] = 1
```

The two loops are represented as follows:

- After `DV`, either `FC` can occur or `RDS` can occur. `RDS` returns the token to the state before document verification.
- After `FC`, either `CA` can occur or `II` can occur. `II` is a self-loop and can therefore be executed repeatedly before proceeding to `CA`.

If approval should be mandatory for every claim, remove the `SKIP_AP` transition and its two arcs.
