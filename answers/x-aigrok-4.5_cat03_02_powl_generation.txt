import pm4py
from pm4py.objects.powl.obj import StrictPartialOrder, OperatorPOWL, Transition, SilentTransition
from pm4py.objects.process_tree.obj import Operator

# Define all atomic activities
SR = Transition(label="SR")
CS = Transition(label="CS")
DA = Transition(label="DA")
ACI = Transition(label="ACI")
BNC = Transition(label="BNC")
DR = Transition(label="D&R")
CD = Transition(label="CD")
MD = Transition(label="MD")
QA = Transition(label="QA")
SA = Transition(label="SA")
GLA = Transition(label="GLA")

# Loop: BNC, then optionally D&R then BNC again (until exit)
loop = OperatorPOWL(operator=Operator.LOOP, children=[BNC, DR])

# Exclusive choice: Containerized or Monolithic deployment
xor = OperatorPOWL(operator=Operator.XOR, children=[CD, MD])

# Root partial order encapsulating the whole process
root = StrictPartialOrder(nodes=[SR, CS, DA, ACI, loop, xor, QA, SA, GLA])

# Ordering constraints
root.order.add_edge(SR, CS)          # verify request then contract
root.order.add_edge(CS, DA)          # contract before datacenter allocation
root.order.add_edge(CS, ACI)         # contract before access-control init
root.order.add_edge(DA, loop)        # both DA & ACI must finish before config loop
root.order.add_edge(ACI, loop)
root.order.add_edge(loop, xor)       # successful configuration before deployment choice
root.order.add_edge(xor, QA)         # deployment before QA
root.order.add_edge(xor, SA)         # deployment before SA
root.order.add_edge(QA, GLA)         # both QA & SA before go-live
root.order.add_edge(SA, GLA)
