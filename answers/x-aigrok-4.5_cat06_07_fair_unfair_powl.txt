```python
import pm4py
from pm4py.objects.powl.obj import StrictPartialOrder, OperatorPOWL, Transition, SilentTransition
from pm4py.objects.process_tree.obj import Operator

# Shared activity definitions
receive = Transition(label="ReceiveApplication")
data_check = Transition(label="DataCompletenessCheck")
request_info = Transition(label="RequestMoreInfo")
skill = Transition(label="SkillAssessment")
cultural = Transition(label="CulturalFitCheck")
community = Transition(label="CommunityAffiliationCheck")
manager = Transition(label="ManagerialReview")
final = Transition(label="FinalDecision")

# Loop for data completeness (common to both models)
loop = OperatorPOWL(operator=Operator.LOOP, children=[data_check, request_info])

# ============================================================
# 1. POWL Model WITH Potential Unfairness
#    (XOR branch containing CommunityAffiliationCheck)
# ============================================================
xor_unfair = OperatorPOWL(operator=Operator.XOR, children=[cultural, community])

root_unfair = StrictPartialOrder(nodes=[receive, loop, skill, xor_unfair, manager, final])
root_unfair.order.add_edge(receive, loop)
root_unfair.order.add_edge(loop, skill)
root_unfair.order.add_edge(skill, xor_unfair)
root_unfair.order.add_edge(xor_unfair, manager)
root_unfair.order.add_edge(manager, final)

# ============================================================
# 2. POWL Model WITHOUT Unfairness
#    (No special affiliation branch; every candidate follows the
#     identical CulturalFitCheck path)
# ============================================================
root_fair = StrictPartialOrder(nodes=[receive, loop, skill, cultural, manager, final])
root_fair.order.add_edge(receive, loop)
root_fair.order.add_edge(loop, skill)
root_fair.order.add_edge(skill, cultural)
root_fair.order.add_edge(cultural, manager)
root_fair.order.add_edge(manager, final)
```
