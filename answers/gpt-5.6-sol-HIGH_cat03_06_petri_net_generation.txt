```python
from pm4py.objects.petri_net.obj import PetriNet, Marking
from pm4py.objects.petri_net.utils import petri_utils


# ------------------------------------------------------------------
# Create the Petri net
# ------------------------------------------------------------------
net = PetriNet(name="Insurance Claim Management")


def add_place(name):
    place = PetriNet.Place(name=name)
    net.places.add(place)
    return place


def add_transition(name, label=None):
    transition = PetriNet.Transition(name=name, label=label)
    net.transitions.add(transition)
    return transition


def add_arc(source, target):
    petri_utils.add_arc_from_to(source, target, net)


# ------------------------------------------------------------------
# Places: states of the claim process
# ------------------------------------------------------------------
p_start = add_place("p_start")
p_claim_filed = add_place("p_claim_filed")
p_before_document_verification = add_place("p_before_document_verification")
p_document_verification_result = add_place("p_document_verification_result")
p_before_fraud_check = add_place("p_before_fraud_check")
p_fraud_check_result = add_place("p_fraud_check_result")
p_before_claim_assessment = add_place("p_before_claim_assessment")
p_claim_assessed = add_place("p_claim_assessed")
p_ready_for_payment = add_place("p_ready_for_payment")
p_paid = add_place("p_paid")
p_notified = add_place("p_notified")
p_closed = add_place("p_closed")


# ------------------------------------------------------------------
# Visible activity transitions
# ------------------------------------------------------------------
t_claim = add_transition(
    "t_claim",
    "Insurance Claim (C)"
)

t_claim_registration = add_transition(
    "t_claim_registration",
    "Claim Registration (CR)"
)

t_document_verification = add_transition(
    "t_document_verification",
    "Document Verification (DV)"
)

t_redocument_submission = add_transition(
    "t_redocument_submission",
    "Re-Document Submission (RDS)"
)

t_fraud_check = add_transition(
    "t_fraud_check",
    "Fraud Check (FC)"
)

t_in_depth_investigation = add_transition(
    "t_in_depth_investigation",
    "In-depth Investigation (II)"
)

t_claim_assessment = add_transition(
    "t_claim_assessment",
    "Claim Assessment (CA)"
)

t_approval = add_transition(
    "t_approval",
    "Approval (AP)"
)

t_payment = add_transition(
    "t_payment",
    "Payment (P)"
)

t_notification = add_transition(
    "t_notification",
    "Notification (N)"
)

t_closure = add_transition(
    "t_closure",
    "Closure (CL)"
)


# ------------------------------------------------------------------
# Invisible routing transitions
#
# A transition with label=None is silent. These transitions represent
# decisions rather than business activities.
# ------------------------------------------------------------------
t_documents_complete = add_transition(
    "t_documents_complete",
    None
)

t_fraud_cleared = add_transition(
    "t_fraud_cleared",
    None
)

t_approval_not_required = add_transition(
    "t_approval_not_required",
    None
)


# ------------------------------------------------------------------
# Main flow:
# start -> C -> CR -> DV
# ------------------------------------------------------------------
add_arc(p_start, t_claim)
add_arc(t_claim, p_claim_filed)

add_arc(p_claim_filed, t_claim_registration)
add_arc(t_claim_registration, p_before_document_verification)

add_arc(p_before_document_verification, t_document_verification)
add_arc(t_document_verification, p_document_verification_result)


# ------------------------------------------------------------------
# Document re-submission loop:
#
# DV result -> RDS -> before DV
#
# After re-submission, the documents are verified again. The loop may
# be repeated until the documents are considered complete.
# ------------------------------------------------------------------
add_arc(p_document_verification_result, t_redocument_submission)
add_arc(t_redocument_submission, p_before_document_verification)

# Documents are complete, so continue to the fraud check.
add_arc(p_document_verification_result, t_documents_complete)
add_arc(t_documents_complete, p_before_fraud_check)


# ------------------------------------------------------------------
# Fraud investigation loop:
#
# FC result -> II -> before FC
#
# After an in-depth investigation, the fraud check is performed again.
# This loop can execute multiple times.
# ------------------------------------------------------------------
add_arc(p_before_fraud_check, t_fraud_check)
add_arc(t_fraud_check, p_fraud_check_result)

add_arc(p_fraud_check_result, t_in_depth_investigation)
add_arc(t_in_depth_investigation, p_before_fraud_check)

# Fraud concerns are cleared, so continue to claim assessment.
add_arc(p_fraud_check_result, t_fraud_cleared)
add_arc(t_fraud_cleared, p_before_claim_assessment)


# ------------------------------------------------------------------
# Claim assessment and conditional approval
# ------------------------------------------------------------------
add_arc(p_before_claim_assessment, t_claim_assessment)
add_arc(t_claim_assessment, p_claim_assessed)

# High-value claim: approval is required.
add_arc(p_claim_assessed, t_approval)
add_arc(t_approval, p_ready_for_payment)

# Lower-value claim: approval is skipped through a silent transition.
add_arc(p_claim_assessed, t_approval_not_required)
add_arc(t_approval_not_required, p_ready_for_payment)


# ------------------------------------------------------------------
# Payment -> notification -> closure
# ------------------------------------------------------------------
add_arc(p_ready_for_payment, t_payment)
add_arc(t_payment, p_paid)

add_arc(p_paid, t_notification)
add_arc(t_notification, p_notified)

add_arc(p_notified, t_closure)
add_arc(t_closure, p_closed)


# ------------------------------------------------------------------
# Initial and final markings
# ------------------------------------------------------------------
im = Marking()
im[p_start] = 1

fm = Marking()
fm[p_closed] = 1


# PM4Py commonly represents an accepting Petri net using these three
# objects:
accepting_petri_net = (net, im, fm)
```

The silent transitions model routing decisions:

- `t_documents_complete`: documents are complete, so RDS is not needed.
- `t_fraud_cleared`: no further investigation is required.
- `t_approval_not_required`: the assessed amount is below the approval threshold.

The final marking is reached only after payment, notification, and closure have occurred.
