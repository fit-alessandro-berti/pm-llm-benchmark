import pm4py
from pm4py.objects.powl.obj import StrictPartialOrder, OperatorPOWL, Transition, SilentTransition
from pm4py.objects.process_tree.obj import Operator

# 1. Initial verification and contract setup
SR = Transition(label="SR")   # Service Request verification
CS = Transition(label="CS")   # Contract Setup

# 2. Parallel network environment preparation (no ordering between them)
DA = Transition(label="DA")   # Datacenter Allocation
ACI = Transition(label="ACI") # Access Control Initialization

# 3. Looped configuration phase: BNC then optionally D&R and repeat
BNC = Transition(label="BNC") # Base Network Configuration
DR = Transition(label="D&R")  # Debug & Reconfigure
loop_node = OperatorPOWL(operator=Operator.LOOP, children=[BNC, DR])

# 4. Exclusive choice for deployment method
CD = Transition(label="CD")   # Containerized Deployment
MD = Transition(label="MD")   # Monolithic Deployment
deploy_choice = OperatorPOWL(operator=Operator.XOR, children=[CD, MD])

# 5. Final parallel quality and security checks (concurrent)
QA = Transition(label="QA")   # Quality Assurance
SA = Transition(label="SA")   # Security Audit

# 6. Final sign-off after both QA and SA are done
GLA = Transition(label="GLA") # Go-Live Approval

# Root partial order assembling all nodes with the required dependencies
root = StrictPartialOrder(nodes=[
    SR, CS, DA, ACI, loop_node, deploy_choice, QA, SA, GLA
])

# SR must complete before CS
root.order.add_edge(SR, CS)

# CS is prerequisite for the parallel preparation tasks
root.order.add_edge(CS, DA)
root.order.add_edge(CS, ACI)

# Both DA and ACI must finish before the loop can start
root.order.add_edge(DA, loop_node)
root.order.add_edge(ACI, loop_node)

# Loop completes before the deployment choice is made
root.order.add_edge(loop_node, deploy_choice)

# Deployment choice completes before QA and SA can begin
root.order.add_edge(deploy_choice, QA)
root.order.add_edge(deploy_choice, SA)

# Both QA and SA must finish before GLA; QA and SA have no ordering between them
root.order.add_edge(QA, GLA)
root.order.add_edge(SA, GLA)

# The resulting POWL model is stored in root
# (No silent transitions are required in this scenario,
