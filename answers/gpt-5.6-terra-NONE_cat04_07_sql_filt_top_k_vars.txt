```sql
-- Replace ? with the desired value of K
WITH case_sequences AS (
    -- Build one ordered activity sequence per case
    SELECT
        case_id,
        string_agg(activity, ' -> ' ORDER BY timestamp) AS activity_sequence
    FROM event_log
    GROUP BY case_id
),

variant_counts AS (
    -- Count how many cases follow each complete sequence
    SELECT
        activity_sequence,
        COUNT(*) AS case_count
    FROM case_sequences
    GROUP BY activity_sequence
),

top_variants AS (
    -- Select the K most frequent variants
    SELECT
        activity_sequence,
        case_count
    FROM variant_counts
    ORDER BY case_count DESC, activity_sequence
    LIMIT ?
),

top_variant_cases AS (
    -- Find all cases whose sequence is one of the top-K variants
    SELECT
        cs.case_id,
        cs.activity_sequence,
        tv.case_count AS variant_case_count
    FROM case_sequences cs
    INNER JOIN top_variants tv
        ON cs.activity_sequence = tv.activity_sequence
)

-- Return original events only for cases in top-K variants
SELECT
    e.case_id,
    e.activity,
    e.timestamp,
    tvc.activity_sequence,
    tvc.variant_case_count
FROM event_log e
INNER JOIN top_variant_cases tvc
    ON e.case_id = tvc.case_id
ORDER BY
    e.case_id,
    e.timestamp;
```

If activities can share the same timestamp within a case, add a deterministic tie-breaker to the ordering, such as an event identifier:

```sql
string_agg(activity, ' -> ' ORDER BY timestamp, event_id)
```

Likewise, use that same ordering in the final `ORDER BY` clause.
