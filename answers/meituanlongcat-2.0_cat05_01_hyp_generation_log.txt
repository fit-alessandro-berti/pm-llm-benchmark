### Proposed SQL Queries (No Hints)

#### 1. Find All Cases with Sequence Violations
```sql
WITH activity_order AS (
    SELECT 'Register Order' AS activity, 1 AS step_pos UNION ALL
    SELECT 'Perform Credit Check', 2 UNION ALL
    SELECT 'Validate Stock', 3 UNION ALL
    SELECT 'Confirm Shipment', 4 UNION ALL
    SELECT 'Ship Goods', 5 UNION ALL
    SELECT 'Issue Invoice', 6 UNION ALL
    SELECT 'Receive Payment', 7
),
ranked_events AS (
    SELECT oel.case_id, oel.activity,
           ROW_NUMBER() OVER (PARTITION BY oel.case_id ORDER BY oel.timestamp, oel.event_id) AS actual_pos
    FROM order_event_log oel
)
SELECT r.case_id, r.activity, r.actual_pos, a.step_pos
FROM ranked_events r
LEFT JOIN activity_order a ON r.activity = a.activity
WHERE a.step_pos IS NOT NULL AND r.actual_pos <> a.step_pos
ORDER BY r.case_id, r.actual_pos;
```

#### 2. Detect Cases with Missing Required Activities
```sql
WITH expected_steps AS (
    SELECT UNNEST(ARRAY[
        'Register Order', 'Perform Credit Check', 'Validate Stock',
        'Confirm Shipment', 'Ship Goods', 'Issue Invoice', 'Receive Payment'
    ]) AS required_activity
),
case_activities AS (
    SELECT DISTINCT case_id, activity
    FROM order_event_log
)
SELECT e.required_activity,
       c.case_id
FROM expected_steps e
CROSS JOIN (SELECT DISTINCT case_id FROM order_event_log) c
LEFT JOIN case_activities ca ON c.case_id = ca.case_id AND e.required_activity = ca.activity
WHERE ca.activity IS NULL
ORDER BY c.case_id, e.required_activity;
```

#### 3. Find Cases Where Ship Goods Preceded Confirm Shipment
```sql
SELECT s.case_id, s.timestamp AS ship_goods_ts, c.timestamp AS confirm_shipment_ts
FROM order_event_log s
JOIN order_event_log c ON s.case_id = c.case_id
WHERE s.activity = 'Ship Goods'
  AND c.activity = 'Confirm Shipment'
  AND s.timestamp < c.timestamp;
```

#### 4. Cases Where Payment Occurred Before Invoice
```sql
SELECT s.case_id, s.timestamp AS payment_ts, i.timestamp AS invoice_ts
FROM order_event_log s
JOIN order_event_log i ON s.case_id = i.case_id
WHERE s.activity = 'Receive Payment'
  AND i.activity = 'Issue Invoice'
  AND s.timestamp < i.timestamp;
```

#### 5. Identify Negative/Flagged Additional Info Indicators
```sql
SELECT case_id, event_id, activity, additional_info
FROM order_event_log
WHERE additional_info IN ('late_confirmation=Y', 'attempted_early=Y', 'shipment_scheduled=N')
ORDER BY case_id, event_id;
```

#### 6. Check Correlation: Priority Orders vs. Sequence Violations
```sql
WITH activity_order AS (
    SELECT 'Register Order' AS activity, 1 AS step_pos UNION ALL
    SELECT 'Perform Credit Check', 2 UNION ALL
    SELECT 'Validate Stock', 3 UNION ALL
    SELECT 'Confirm Shipment', 4 UNION ALL
    SELECT 'Ship Goods', 5 UNION ALL
    SELECT 'Issue Invoice', 6 UNION ALL
    SELECT 'Receive Payment', 7
),
ranked AS (
    SELECT oel.case_id, oel.activity,
           ROW_NUMBER() OVER (PARTITION BY oel.case_id ORDER BY oel.timestamp, oel.event_id) AS actual_pos
    FROM order_event_log oel
)
SELECT r.case_id, o.order_type, r.activity,
       r.actual_pos, a.step_pos,
       CASE WHEN r.actual_pos <> a.step_pos THEN 'VIOLATION' ELSE 'OK' END AS status
FROM ranked r
JOIN orders o ON r.case_id = o.case_id
JOIN activity_order a ON r.activity = a.activity
ORDER BY r.case_id, r.actual_pos;
```

#### 7. Detect Cases Where Credit Check Occurred After Shipment
```sql
SELECT cc.case_id, cc.timestamp AS credit_check_ts, sg.timestamp AS ship_goods_ts
FROM order_event_log cc
JOIN order_event_log sg ON cc.case_id = sg.case_id
WHERE cc.activity = 'Perform Credit Check'
  AND sg.activity = 'Ship Goods'
  AND cc.timestamp > sg.timestamp;
```

#### 8. Cases with Only a Subset of Activities (Potential Abandonment or Bypass)
```sql
SELECT case_id,
       STRING_AGG(activity, '' ORDER BY timestamp, event_id) AS actual_flow,
       COUNT(*) AS activity_count
FROM order_event_log
GROUP BY case_id
HAVING COUNT(*) < 7
ORDER BY case_id;
```

#### 9. Activity-to-Department Mismatch Verification
```sql
WITH expected_dept AS (
    SELECT 'Register Order' AS activity, 'Sales' AS expected_dept UNION ALL
    SELECT 'Perform Credit Check', 'Finance' UNION ALL
    SELECT 'Validate Stock', 'Logistics' UNION ALL
    SELECT 'Confirm Shipment', 'Logistics' UNION ALL
    SELECT 'Ship Goods', 'Logistics' UNION ALL
    SELECT 'Issue Invoice', 'Finance' UNION ALL
    SELECT 'Receive Payment', 'Finance'
)
SELECT oel.case_id, oel.activity, oel.resource,
       r.department, r.role,
       e.expected_dept,
       CASE WHEN r.department <> e.expected_dept THEN 'MISMATCH' ELSE 'OK' END AS dept_check
FROM order_event_log oel
JOIN resources r ON oel.resource = r.resource_id
JOIN expected_dept e ON oel.activity = e.activity
WHERE r.department <> e.expected_dept
ORDER BY oel.case_id, oel.event_id;
```

#### 10. Concurrency Check: Events with Timestamps Within 2 Minutes
```sql
SELECT a.case_id, a.event_id, a.activity, a.timestamp,
       b.event_id, b.activity, b.timestamp,
       EXTRACT(EPOCH FROM (b.timestamp - a.timestamp))/60 AS min_diff
FROM order_event_log a
JOIN order_event_log b ON a.case_id = b.case_id AND a.event_id < b.event_id
WHERE ABS(EXTRACT(EPOCH FROM (b.timestamp - a.timestamp))/60) < 2
ORDER BY a.case_id, a.event_id;
```
