```python
from pm4py.objects.petri_net.obj import PetriNet, Marking
from pm4py.objects.petri_net.utils import petri_utils

# Create the Petri net
net = PetriNet(name="insurance_claim_management")

# Places (states between activities)
start = PetriNet.Place("start")                  # initial place (new claim arrives)
p_after_C = PetriNet.Place("p_after_C")          # after Insurance Claim (C)
p_after_CR = PetriNet.Place("p_after_CR")        # after Claim Registration (CR) / before DV
p_after_DV = PetriNet.Place("p_after_DV")        # after Document Verification (DV) – decision point
p_before_FC = PetriNet.Place("p_before_FC")      # before Fraud Check (FC)
p_after_FC = PetriNet.Place("p_after_FC")        # after Fraud Check (FC) – decision point
p_before_CA = PetriNet.Place("p_before_CA")      # before Claim Assessment (CA)
p_after_CA = PetriNet.Place("p_after_CA")        # after Claim Assessment (CA)
p_after_AP = PetriNet.Place("p_after_AP")        # after Approval (AP)
p_after_P = PetriNet.Place("p_after_P")          # after Payment (P)
p_after_N = PetriNet.Place("p_after_N")          # after Notification (N)
end = PetriNet.Place("end")                      # final place (claim closed)

# Add all places to the net
for place in [start, p_after_C, p_after_CR, p_after_DV, p_before_FC,
              p_after_FC, p_before_CA, p_after_CA, p_after_AP,
              p_after_P, p_after_N, end]:
    net.places.add(place)

# Transitions (activities) – silent transitions (label=None) model exclusive choices
t_C = PetriNet.Transition(name="C", label="Insurance Claim (C)")
t_CR = PetriNet.Transition(name="CR", label="Claim Registration (CR)")
t_DV = PetriNet.Transition(name="DV", label="Document Verification (DV)")
t_RDS = PetriNet.Transition(name="RDS", label="Re-Document Submission (RDS)")
t_tau_DV = PetriNet.Transition(name="tau_proceed_after_DV", label=None)  # proceed to FC
t_FC = PetriNet.Transition(name="FC", label="Fraud Check (FC)")
t_II = PetriNet.Transition(name="II", label="In-depth Investigation (II)")
t_tau_FC = PetriNet.Transition(name="tau_proceed_after_FC", label=None)  # proceed to CA
t_CA = PetriNet.Transition(name="CA", label="Claim Assessment (CA)")
t_AP = PetriNet.Transition(name="AP", label="Approval (AP)")
t_P = PetriNet.Transition(name="P", label="Payment (P)")
t_N = PetriNet.Transition(name="N", label="Notification (N)")
t_CL = PetriNet.Transition(name="CL", label="Closure (CL)")

# Add all transitions to the net
for transition in [t_C, t_CR, t_DV, t_RDS, t_tau_DV, t_FC, t_II,
                   t_tau_FC, t_CA, t_AP, t_P, t_N, t_CL]:
    net.transitions.add(transition)

# Arcs – sequential flow with loops for re-submission and investigation
# Start → C → CR → (DV loop)
petri_utils.add_arc_from_to(start, t_C, net)
petri_utils.add_arc_from_to(t_C, p_after_C, net)
petri_utils.add_arc_from_to(p_after_C, t_CR, net)
petri_utils.add_arc_from_to(t_CR, p_after_CR, net)

# DV ↔ RDS loop (documents may be incomplete → re-submit → verify again)
petri_utils.add_arc_from_to(p_after_CR, t_DV, net)
petri_utils.add_arc_from_to(t_DV, p_after_DV, net)
petri_utils.add_arc_from_to(p_after_DV, t_RDS, net)
petri_utils.add_arc_from_to(t_RDS, p_after_CR, net)          # back to before DV
petri_utils.add_arc_from_to(p_after_DV, t_tau_DV, net)       # proceed when documents OK
petri_utils.add_arc_from_to(t_tau_DV, p_before_FC, net)

# FC ↔ II loop (fraud doubts → investigate → re-check, possibly multiple times)
petri_utils.add_arc_from_to(p_before_FC, t_FC, net)
petri_utils.add_arc_from_to(t_FC, p_after_FC, net)
petri_utils.add_arc_from_to(p_after_FC, t_II, net)
petri_utils.add_arc_from_to(t_II, p_before_FC, net)          # back to before FC
petri_utils.add_arc_from_to(p_after_FC, t_tau_FC, net)       # proceed when no more doubts
petri_utils.add_arc_from_to(t_tau_FC, p_before_CA, net)

# CA → AP → P → N → CL → end
petri_utils.add_arc_from_to(p_before_CA, t_CA, net)
petri_utils.add_arc_from_to(t_CA, p_after_CA, net)
petri_utils.add_arc_from_to(p_after_CA, t_AP, net)
petri_utils.add_arc_from_to(t_AP, p_after_AP, net)
petri_utils.add_arc_from_to(p_after_AP, t_P, net)
petri_utils.add_arc_from_to(t_P, p_after_P, net)
petri_utils.add_arc_from_to(p_after_P, t_N, net)
petri_utils.add_arc_from_to(t_N, p_after_N, net)
petri_utils.add_arc_from_to(p_after_N, t_CL, net)
petri_utils.add_arc_from_to(t_CL, end, net)

# Initial marking: one token in the start place (new claim arrives)
im = Marking()
im[start] = 1

# Final marking: one token in the end place (claim closed and archived)
fm = Marking()
fm[end] = 1
```
