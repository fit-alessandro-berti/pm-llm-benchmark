MS  = Transition(label="MS")
QS  = Transition(label="QS")
AS  = Transition(label="AS")
IQC = Transition(label="IQC")
TST = Transition(label="TST")

loop_assembly_qc = OperatorPOWL(operator=Operator.LOOP, children=[AS, loop_iqc])

# 3. Testing / re-testing loop
loop_test = OperatorPOWL(operator=Operator.LOOP, children=[TST, RT])

# 4. Labeling is optional
choice_label = OperatorPOWL(operator=Operator.XOR, children=[LB, skip])
