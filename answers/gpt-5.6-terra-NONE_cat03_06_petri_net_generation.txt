```python
from pm4py.objects.petri_net.obj import PetriNet, Marking
from pm4py.objects.petri_net.utils import petri_utils

# Create the Petri net
net = PetriNet(name="Insurance_Claim_Management")

# -------------------------------------------------------------------
# Places: represent the states between activities
# -------------------------------------------------------------------
p_start = PetriNet.Place("start")
p_after_claim = PetriNet.Place("after_claim")
p_after_registration = PetriNet.Place("after_registration")
p_after_dv = PetriNet.Place("after_document_verification")
p_after_rds = PetriNet.Place("after_re_document_submission")
p_after_fc = PetriNet.Place("after_fraud_check")
p_after_ii = PetriNet.Place("after_in_depth_investigation")
p_after_ca = PetriNet.Place("after_claim_assessment")
p_after_ap = PetriNet.Place("after_approval")
p_after_payment = PetriNet.Place("after_payment")
p_after_notification = PetriNet.Place("after_notification")
p_closed = PetriNet.Place("closed")

for place in [
    p_start,
    p_after_claim,
    p_after_registration,
    p_after_dv,
    p_after_rds,
    p_after_fc,
    p_after_ii,
    p_after_ca,
    p_after_ap,
    p_after_payment,
    p_after_notification,
    p_closed
]:
    net.places.add(place)

# -------------------------------------------------------------------
# Transitions: represent process activities
# -------------------------------------------------------------------
t_claim = PetriNet.Transition("C", "Insurance Claim")
t_registration = PetriNet.Transition("CR", "Claim Registration")
t_document_verification = PetriNet.Transition("DV", "Document Verification")
t_re_document_submission = PetriNet.Transition("RDS", "Re-Document Submission")
t_fraud_check = PetriNet.Transition("FC", "Fraud Check")
t_in_depth_investigation = PetriNet.Transition("II", "In-depth Investigation")
t_claim_assessment = PetriNet.Transition("CA", "Claim Assessment")
t_approval = PetriNet.Transition("AP", "Approval")
t_payment = PetriNet.Transition("P", "Payment")
t_notification = PetriNet.Transition("N", "Notification")
t_closure = PetriNet.Transition("CL", "Closure")

for transition in [
    t_claim,
    t_registration,
    t_document_verification,
    t_re_document_submission,
    t_fraud_check,
    t_in_depth_investigation,
    t_claim_assessment,
    t_approval,
    t_payment,
    t_notification,
    t_closure
]:
    net.transitions.add(transition)

# -------------------------------------------------------------------
# Main process flow
# -------------------------------------------------------------------
petri_utils.add_arc_from_to(p_start, t_claim, net)
petri_utils.add_arc_from_to(t_claim, p_after_claim, net)

petri_utils.add_arc_from_to(p_after_claim, t_registration, net)
petri_utils.add_arc_from_to(t_registration, p_after_registration, net)

petri_utils.add_arc_from_to(p_after_registration, t_document_verification, net)
petri_utils.add_arc_from_to(t_document_verification, p_after_dv, net)

# -------------------------------------------------------------------
# Document re-submission loop:
# DV -> RDS -> DV
# After DV, the token can either continue to Fraud Check or
# go through Re-Document Submission and return to DV.
# -------------------------------------------------------------------
petri_utils.add_arc_from_to(p_after_dv, t_re_document_submission, net)
petri_utils.add_arc_from_to(t_re_document_submission, p_after_rds, net)
petri_utils.add_arc_from_to(p_after_rds, t_document_verification, net)

# Continue after successful document verification
petri_utils.add_arc_from_to(p_after_dv, t_fraud_check, net)
petri_utils.add_arc_from_to(t_fraud_check, p_after_fc, net)

# -------------------------------------------------------------------
# In-depth investigation loop:
# FC -> II -> FC
# Fraud checks can be repeated after an investigation.
# -------------------------------------------------------------------
petri_utils.add_arc_from_to(p_after_fc, t_in_depth_investigation, net)
petri_utils.add_arc_from_to(t_in_depth_investigation, p_after_ii, net)
petri_utils.add_arc_from_to(p_after_ii, t_fraud_check, net)

# Continue to claim assessment after fraud check is cleared
petri_utils.add_arc_from_to(p_after_fc, t_claim_assessment, net)
petri_utils.add_arc_from_to(t_claim_assessment, p_after_ca, net)

# Approval, payment, notification, and closure
petri_utils.add_arc_from_to(p_after_ca, t_approval, net)
petri_utils.add_arc_from_to(t_approval, p_after_ap, net)

petri_utils.add_arc_from_to(p_after_ap, t_payment, net)
petri_utils.add_arc_from_to(t_payment, p_after_payment, net)

petri_utils.add_arc_from_to(p_after_payment, t_notification, net)
petri_utils.add_arc_from_to(t_notification, p_after_notification, net)

petri_utils.add_arc_from_to(p_after_notification, t_closure, net)
petri_utils.add_arc_from_to(t_closure, p_closed, net)

# -------------------------------------------------------------------
# Initial and final markings
# -------------------------------------------------------------------
im = Marking()
im[p_start] = 1

fm = Marking()
fm[p_closed] = 1
```

The resulting accepting Petri net starts with one token in `p_start` and reaches the final marking when one token arrives in `p_closed` after the `Closure` transition. The `RDS` and `II` transitions create repeatable loops for incomplete documentation and repeated fraud investigations, respectively.
