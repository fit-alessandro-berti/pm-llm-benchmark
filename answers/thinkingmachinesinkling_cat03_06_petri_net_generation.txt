Here is the complete Python code that builds the accepting Petri net for the insurance claim process, including the document-re-submission loop (`DV`  `RDS`) and the investigation loop (`FC`  `II`).

```python
from pm4py.objects.petri_net.obj import PetriNet, Marking
from pm4py.objects.petri_net.utils import petri_utils

# 1. Create the Petri net
net = PetriNet(name="insurance_claim_management")

# 2. Define places (states between activities)
start      = PetriNet.Place(name="start")      # new claim arrived
after_CR   = PetriNet.Place(name="after_CR")   # claim registered
after_DV   = PetriNet.Place(name="after_DV")   # documents verified / ready for fraud check
after_FC   = PetriNet.Place(name="after_FC")   # fraud check completed
after_CA   = PetriNet.Place(name="after_CA")   # claim assessed
after_AP   = PetriNet.Place(name="after_AP")   # claim approved
after_P    = PetriNet.Place(name="after_P")    # payment made
after_N    = PetriNet.Place(name="after_N")    # notification sent
sink       = PetriNet.Place(name="sink")       # claim closed / archived

# Add places to the net
for p in [start, after_CR, after_DV, after_FC, after_CA, after_AP, after_P, after_N, sink]:
    net.places.add(p)

# 3. Define transitions (activities)
cr = PetriNet.Transition(name="CR", label="Claim Registration")
dv = PetriNet.Transition(name="DV", label="Document Verification")
rds = PetriNet.Transition(name="RDS", label="Re-Document Submission")
fc = PetriNet.Transition(name="FC", label="Fraud Check")
ii = PetriNet.Transition(name="II", label="In-depth Investigation")
ca = PetriNet.Transition(name="CA", label="Claim Assessment")
ap = PetriNet.Transition(name="AP", label="Approval")
p_trans = PetriNet.Transition(name="P", label="Payment")
n = PetriNet.Transition(name="N", label="Notification")
cl = PetriNet.Transition(name="CL", label="Closure")

# Add transitions to the net
for t in [cr, dv, rds, fc, ii, ca, ap, p_trans, n, cl]:
    net.transitions.add(t)

# 4. Connect places and transitions with arcs

# Main linear flow
petri_utils.add_arc_from_to(start, cr, net)
petri_utils.add_arc_from_to(cr, after_CR, net)

petri_utils.add_arc_from_to(after_CR, dv, net)
petri_utils.add_arc_from_to(dv, after_DV, net)

# Document Verification loop: if incomplete -> RDS -> back to DV input
petri_utils.add_arc_from_to(after_DV, rds, net)
petri_utils.add_arc_from_to(rds, after_CR, net)   # loop back before DV

# Fraud Check loop: FC -> (clear) or (doubts -> II -> back to FC)
petri_utils.add_arc_from_to(after_DV, fc, net)
petri_utils.add_arc_from_to(fc, after_FC, net)

# From after FC: either proceed to Assessment or loop back via Investigation
petri_utils.add_arc_from_to(after_FC, ca, net)
petri_utils.add_arc_from_to(after_FC, ii, net)
petri_utils.add_arc_from_to(ii, after_DV, net)   # loop back to FC input

# Remaining sequential steps
petri_utils.add_arc_from_to(ca, after_CA, net)
petri_utils.add_arc_from_to(after_CA, ap, net)
petri_utils.add_arc_from_to(ap, after_AP, net)
petri_utils.add_arc_from_to(after_AP, p_trans, net)
petri_utils.add_arc_from_to(p_trans, after_P, net)
petri_utils.add_arc_from_to(after_P, n, net)
petri_utils.add_arc_from_to(n, after_N, net)
petri_utils.add_arc_from_to(after_N, cl, net)
petri_utils.add_arc_from_to(cl, sink, net)

# 5. Define initial and final markings
im = Marking()
im[start] = 1

fm = Marking()
fm[sink] = 1
```

**Key design points:**
- **`DV` loop:** `after_DV` branches to `RDS`, which returns the token to `after_CR` so `DV` can be executed again.
- **`FC` loop:** `after_FC` branches to `II`, which returns the token to `after_DV` so `FC` can be re-executed (potentially multiple times).
- **Linear tail:** Once `CA` succeeds, the flow proceeds strictly through `AP`  `P`  `N`  `CL`  `sink`.
- **Markings:** `im` puts one token in `start` (new claim). `fm` puts one token in `sink` (closed claim).
