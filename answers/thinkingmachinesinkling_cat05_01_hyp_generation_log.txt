  STRING_AGG(
    activity || ' [' || resource || ']@' || TO_CHAR(timestamp, 'HH24:MI') || 
    COALESCE(' {' || additional_info || '}', ''),
    ' -> ' ORDER BY timestamp, event_id
  ) AS process_trail
FROM order_event_log
GROUP BY case_id
ORDER BY 
  CASE 
    WHEN case_id IN (SELECT case_id FROM order_event_log GROUP BY case_id HAVING COUNT(DISTINCT activity) < 7) 
    THEN 0 ELSE 1 
  END,
  case_id;
