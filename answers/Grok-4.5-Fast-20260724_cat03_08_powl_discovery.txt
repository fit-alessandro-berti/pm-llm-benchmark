```python
import pm4py
from pm4py.objects.powl.obj import StrictPartialOrder, OperatorPOWL, Transition, SilentTransition
from pm4py.objects.process_tree.obj import Operator

# Atomic activities
OR = Transition(label="OR")
MS = Transition(label="MS")
QS = Transition(label="QS")
AS = Transition(label="AS")
IQC = Transition(label="IQC")
TST = Transition(label="TST")
RT = Transition(label="RT")
PK = Transition(label="PK")
LB = Transition(label="LB")
DP = Transition(label="DP")
D = Transition(label="D")
skip = SilentTransition()

# Zero or more consecutive IQCs: * (tau, IQC)
iqc_star = OperatorPOWL(operator=Operator.LOOP, children=[skip, IQC])

# One AS followed by zero or more IQCs
as_then_iqcs = StrictPartialOrder(nodes=[AS, iqc_star])
as_then_iqcs.order.add_edge(AS, iqc_star)

# At least one (AS + IQC*): * (as_then_iqcs, tau)
assembly = OperatorPOWL(operator=Operator.LOOP, children=[as_then_iqcs, skip])

# Testing with optional re-tests: * (TST, RT) = TST (RT TST)*
test_loop = OperatorPOWL(operator=Operator.LOOP, children=[TST, RT])

# Optional labeling
lb_choice = OperatorPOWL(operator=Operator.XOR, children=[LB, skip])

# Packaging sequence: PK → (LB | tau) → DP → D
packaging = StrictPartialOrder(nodes=[PK, lb_choice, DP, D])
packaging.order.add_edge(PK, lb_choice)
packaging.order.add_edge(lb_choice, DP)
packaging.order.add_edge(DP, D)

# Overall sequential process
root = StrictPartialOrder(nodes=[OR, MS, QS, assembly, test_loop, packaging])
root.order.add_edge(OR, MS)
root.order.add_edge(MS, QS)
root.order.add_edge(QS, assembly)
root.order.add_edge(assembly, test_loop)
root.order.add_edge(test_loop, packaging)
```
