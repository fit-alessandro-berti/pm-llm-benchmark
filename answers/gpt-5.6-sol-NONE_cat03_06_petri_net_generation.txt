```python
from pm4py.objects.petri_net.obj import PetriNet, Marking
from pm4py.objects.petri_net.utils import petri_utils


# Create the Petri net
net = PetriNet("insurance_claim_management")


# ------------------------------------------------------------------
# Places: states between activities
# ------------------------------------------------------------------
start = PetriNet.Place("start")
after_claim = PetriNet.Place("after_claim")
after_registration = PetriNet.Place("after_registration")
after_document_verification = PetriNet.Place("after_document_verification")
after_fraud_check = PetriNet.Place("after_fraud_check")
after_assessment = PetriNet.Place("after_assessment")
after_approval = PetriNet.Place("after_approval")
after_payment = PetriNet.Place("after_payment")
after_notification = PetriNet.Place("after_notification")
closed = PetriNet.Place("closed")

net.places.update({
    start,
    after_claim,
    after_registration,
    after_document_verification,
    after_fraud_check,
    after_assessment,
    after_approval,
    after_payment,
    after_notification,
    closed,
})


# ------------------------------------------------------------------
# Visible activity transitions
# ------------------------------------------------------------------
claim = PetriNet.Transition("C", "Insurance Claim")
registration = PetriNet.Transition("CR", "Claim Registration")
document_verification = PetriNet.Transition("DV", "Document Verification")
document_resubmission = PetriNet.Transition(
    "RDS", "Re-Document Submission"
)
fraud_check = PetriNet.Transition("FC", "Fraud Check")
investigation = PetriNet.Transition("II", "In-depth Investigation")
assessment = PetriNet.Transition("CA", "Claim Assessment")
approval = PetriNet.Transition("AP", "Approval")
payment = PetriNet.Transition("P", "Payment")
notification = PetriNet.Transition("N", "Notification")
closure = PetriNet.Transition("CL", "Closure")

# Invisible transitions model routing decisions.
# None means that they do not correspond to logged activities.
documents_complete = PetriNet.Transition(
    "tau_documents_complete", None
)
fraud_cleared = PetriNet.Transition(
    "tau_fraud_cleared", None
)
approval_not_required = PetriNet.Transition(
    "tau_approval_not_required", None
)

net.transitions.update({
    claim,
    registration,
    document_verification,
    document_resubmission,
    fraud_check,
    investigation,
    assessment,
    approval,
    payment,
    notification,
    closure,
    documents_complete,
    fraud_cleared,
    approval_not_required,
})


# Helper for adding an input and output arc for a transition
def connect(source, transition, target):
    petri_utils.add_arc_from_to(source, transition, net)
    petri_utils.add_arc_from_to(transition, target, net)


# ------------------------------------------------------------------
# Main flow
# ------------------------------------------------------------------
connect(start, claim, after_claim)
connect(after_claim, registration, after_registration)
connect(after_registration, document_verification,
        after_document_verification)

# Document verification outcome:
# 1. Documents are complete: continue to Fraud Check.
# 2. Documents are incomplete: perform RDS and verify them again.
connect(after_document_verification, documents_complete,
        after_fraud_check)
connect(after_document_verification, document_resubmission,
        after_registration)

connect(after_fraud_check, fraud_check, after_assessment)

# Fraud-check outcome:
# 1. Fraud concerns are cleared: proceed to assessment.
# 2. Doubts remain: perform II, then repeat the Fraud Check.
connect(after_assessment, fraud_cleared, after_approval)
connect(after_assessment, investigation, after_fraud_check)

connect(after_approval, assessment, after_payment)

# Approval is either required or skipped based on the amount.
connect(after_payment, approval, after_notification)
connect(after_payment, approval_not_required, after_notification)

connect(after_notification, payment, closed)

# Add separate places for the final notification and closure sequence.
# The names above can be adjusted without changing transition semantics.
payment_done = PetriNet.Place("payment_done")
notification_done = PetriNet.Place("notification_done")
net.places.update({payment_done, notification_done})

# Replace the temporary AP -> notification -> payment -> closed tail.
petri_utils.remove_arc(
    net,
    next(a for a in net.arcs
         if a.source == approval and a.target == after_notification)
)
petri_utils.remove_arc(
    net,
    next(a for a in net.arcs
         if a.source == approval_not_required
         and a.target == after_notification)
)
petri_utils.remove_arc(
    net,
    next(a for a in net.arcs
         if a.source == after_notification and a.target == payment)
)
petri_utils.remove_arc(
    net,
    next(a for a in net.arcs
         if a.source == payment and a.target == closed)
)

petri_utils.add_arc_from_to(approval, payment_done, net)
petri_utils.add_arc_from_to(approval_not_required, payment_done, net)
connect(payment_done, payment, after_notification)
connect(after_notification, notification, notification_done)
connect(notification_done, closure, closed)


# ------------------------------------------------------------------
# Initial and final markings
# ------------------------------------------------------------------
im = Marking()
im[start] = 1

fm = Marking()
fm[closed] = 1
```

The resulting behavior is:

```text
C  CR  DV  (RDS  DV)*  FC  (II  FC)*  CA
   (AP or approval skipped)  P  N  CL
```

`im` contains one token in `start`, while `fm` requires one token in `closed`.
