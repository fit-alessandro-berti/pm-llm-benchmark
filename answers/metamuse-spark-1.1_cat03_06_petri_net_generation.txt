For the insurance claim process the main flow is

`C -> CR -> DV -> FC -> CA -> AP -> P -> N -> CL`

with two loops:

* `DV -> RDS -> DV` if documents are incomplete
* `FC -> II -> FC` if fraud check raises doubts

`AP` is required only above a threshold. It is modelled as optional via a silent `skip_AP` transition.

```python
from pm4py.objects.petri_net.obj import PetriNet, Marking
from pm4py.objects.petri_net.utils import petri_utils

# Create net
net = PetriNet(name="InsuranceClaimManagement")

# --- Places: states between activities ---
p_start = PetriNet.Place(name="p_start") # new claim arrived
p_after_C = PetriNet.Place(name="p_after_C")
p_before_DV = PetriNet.Place(name="p_before_DV") # after CR
p_after_DV = PetriNet.Place(name="p_after_DV_before_FC") # decision after DV
p_after_FC = PetriNet.Place(name="p_after_FC_before_CA") # decision after FC
p_after_CA = PetriNet.Place(name="p_after_CA_before_AP")
p_after_AP = PetriNet.Place(name="p_after_AP_before_P")
p_after_P = PetriNet.Place(name="p_after_P_before_N")
p_after_N = PetriNet.Place(name="p_after_N_before_CL")
p_end = PetriNet.Place(name="p_end") # closed claim

for p in [p_start, p_after_C, p_before_DV, p_after_DV, p_after_FC, 
          p_after_CA, p_after_AP, p_after_P, p_after_N, p_end]:
    net.places.add(p)

# --- Transitions: labelled with activities ---
t_C = PetriNet.Transition(name="C", label="C")
t_CR = PetriNet.Transition(name="CR", label="Claim Registration")
t_DV = PetriNet.Transition(name="DV", label="Document Verification")
t_RDS = PetriNet.Transition(name="RDS", label="Re-Document Submission")
t_FC = PetriNet.Transition(name="FC", label="Fraud Check")
t_II = PetriNet.Transition(name="II", label="In-depth Investigation")
t_CA = PetriNet.Transition(name="CA", label="Claim Assessment")
t_AP = PetriNet.Transition(name="AP", label="Approval")
t_skip_AP = PetriNet.Transition(name="skip_AP", label=None) # silent, AP is optional
t_P = PetriNet.Transition(name="P", label="Payment")
t_N = PetriNet.Transition(name="N", label="Notification")
t_CL = PetriNet.Transition(name="CL", label="Closure")

for t in [t_C, t_CR, t_DV, t_RDS, t_FC, t_II, t_CA, t_AP, t_skip_AP, t_P, t_N, t_CL]:
    net.transitions.add(t)

# --- Arcs ---
# C -> CR -> DV
petri_utils.add_arc_from_to(p_start, t_C, net)
petri_utils.add_arc_from_to(t_C, p_after_C, net)
petri_utils.add_arc_from_to(p_after_C, t_CR, net)
petri_utils.add_arc_from_to(t_CR, p_before_DV, net)
petri_utils.add_arc_from_to(p_before_DV, t_DV, net)
petri_utils.add_arc_from_to(t_DV, p_after_DV, net)

# Loop: DV <-> RDS
petri_utils.add_arc_from_to(p_after_DV, t_RDS, net)
petri_utils.add_arc_from_to(t_RDS, p_before_DV, net)

# FC and Loop: FC <-> II
petri_utils.add_arc_from_to(p_after_DV, t_FC, net)
petri_utils.add_arc_from_to(t_FC, p_after_FC, net)
petri_utils.add_arc_from_to(p_after_FC, t_II, net)
petri_utils.add_arc_from_to(t_II, p_after_DV, net) # re-check after investigation

# CA -> AP -> P -> N -> CL
petri_utils.add_arc_from_to(p_after_FC, t_CA, net)
petri_utils.add_arc_from_to(t_CA, p_after_CA, net)

# optional approval
petri_utils.add_arc_from_to(p_after_CA, t_AP, net)
petri_utils.add_arc_from_to(t_AP, p_after_AP, net)
petri_utils.add_arc_from_to(p_after_CA, t_skip_AP, net)
petri_utils.add_arc_from_to(t_skip_AP, p_after_AP, net)

petri_utils.add_arc_from_to(p_after_AP, t_P, net)
petri_utils.add_arc_from_to(t_P, p_after_P, net)
petri_utils.add_arc_from_to(p_after_P, t_N, net)
petri_utils.add_arc_from_to(t_N, p_after_N, net)
petri_utils.add_arc_from_to(p_after_N, t_CL, net)
petri_utils.add_arc_from_to(t_CL, p_end, net)

# --- Initial and Final Marking ---
im = Marking()
im[p_start] = 1

fm = Marking()
fm[p_end] = 1
```

The net is accepting: `im` puts one token in `p_start` and `fm` requires one token in `p_end` after `Closure`. 

* Token in `p_after_DV` enables either `RDS` to return to `p_before_DV` or `FC` to proceed.
* Token in `p_after_FC` enables either `II` to return to `p_after_DV` for re-checking or `CA` to proceed.
* `skip_AP` allows bypassing `Approval` when the amount is below threshold.
