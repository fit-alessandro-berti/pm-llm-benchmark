The process is fully sequential with two nested loops and one optional step:

* `AS-IQC` : `AS` with optional redo of `1..n IQC` and `AS`  
`IQC+ = *(IQC,IQC)`  - one or more `IQC`  
`Assembly = *(AS, IQC+)` - `Case 4` exits after first `AS`

* `TST-RT` : `*(TST,RT)`  - `TST (RT TST)*`

* `LB` is skippable: `X(LB,tau)` after `PK` and before `DP`

```python
import pm4py
from pm4py.objects.powl.obj import StrictPartialOrder, OperatorPOWL, Transition, SilentTransition
from pm4py.objects.process_tree.obj import Operator

# activities
OR = Transition(label="OR")
MS = Transition(label="MS")
QS = Transition(label="QS")
AS = Transition(label="AS")
IQC = Transition(label="IQC")
TST = Transition(label="TST")
RT = Transition(label="RT")
PK = Transition(label="PK")
LB = Transition(label="LB")
DP = Transition(label="DP")
D = Transition(label="D")
skip = SilentTransition()

# IQC can repeat consecutively: Case3 IQC IQC
# LOOP(IQC,IQC) = IQC+, 1..n times
iqc_one_or_more = OperatorPOWL(operator=Operator.LOOP, children=[IQC, IQC])

# Assembly block: AS -[IQC+-AS]*
# Covers:
# AS alone -> Case 4
# AS IQC AS -> Case 1,2,5
# AS IQC IQC AS IQC AS -> Case 3
assembly_loop = OperatorPOWL(operator=Operator.LOOP, children=[AS, iqc_one_or_more])

# Testing block: TST -[RT-TST]*
# Covers TST alone and TST RT TST ... 
test_loop = OperatorPOWL(operator=Operator.LOOP, children=[TST, RT])

# Labeling may be skipped: Case2
lb_choice = OperatorPOWL(operator=Operator.XOR, children=[LB, skip])

# top level partial order: fully sequential chain
root = StrictPartialOrder(nodes=[OR, MS, QS, assembly_loop, test_loop, PK, lb_choice, DP, D])
root.order.add_edge(OR, MS)
root.order.add_edge(MS, QS)
root.order.add_edge(QS, assembly_loop)
root.order.add_edge(assembly_loop, test_loop)
root.order.add_edge(test_loop, PK)
root.order.add_edge(PK, lb_choice)
root.order.add_edge(lb_choice, DP)
root.order.add_edge(DP, D)
```

Resulting POWL:

```
PO=(nodes={
  OR, MS, QS,
  *( AS, *(IQC, IQC) ),
  *( TST, RT ),
  PK,
  X(LB, tau),
  DP, D
}, order={
  OR-->MS-->QS-->*(AS,*(IQC,IQC))-->*(TST,RT)-->PK-->X(LB,tau)-->DP-->D
})
```
