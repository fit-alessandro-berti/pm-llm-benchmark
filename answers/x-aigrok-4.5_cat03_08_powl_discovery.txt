```python
import pm4py
from pm4py.objects.powl.obj import StrictPartialOrder, OperatorPOWL, Transition, SilentTransition
from pm4py.objects.process_tree.obj import Operator

# Define all activities as transitions
OR = Transition(label="OR")
MS = Transition(label="MS")
QS = Transition(label="QS")
AS = Transition(label="AS")
IQC = Transition(label="IQC")
TST = Transition(label="TST")
RT = Transition(label="RT")
PK = Transition(label="PK")
LB = Transition(label="LB")
DP = Transition(label="DP")
D = Transition(label="D")

# Silent transition for optional paths / multi-instance loops
skip = SilentTransition()

# Loop allowing 1+ consecutive IQC activities (for "repeated if needed")
iqc_loop = OperatorPOWL(operator=Operator.LOOP, children=[IQC, skip])

# Assembly loop: always start with AS, optionally redo via (1+ IQCs followed by re-AS)
# Produces: AS (IQC+ AS)* — matches all observed re-assembly patterns
assembly = OperatorPOWL(operator=Operator.LOOP, children=[AS, iqc_loop])

# Testing loop: always start with TST, optionally re-test via RT
# Produces: TST (RT TST)* — matches all observed pass/fail re-test patterns
test_loop = OperatorPOWL(operator=Operator.LOOP, children=[TST, RT])

# Choice: labeling is optional (can be skipped)
xor_lb = OperatorPOWL(operator=Operator.XOR, children=[LB, skip])

# Top-level partial order enforcing the overall sequential process flow
# (unconnected nodes would be concurrent, but everything here is ordered)
root = StrictPartialOrder(nodes=[OR, MS, QS, assembly, test_loop, PK, xor_lb, DP, D])
root.order.add_edge(OR, MS)
root.order.add_edge(MS, QS)
root.order.add_edge(QS, assembly)
root.order.add_edge(assembly, test_loop)
root.order.add_edge(test_loop, PK)
root.order.add_edge(PK, xor_lb)
root.order.add_edge(xor_lb, DP)
root.order.add_edge(DP, D)
```
