from pm4py.objects.powl.obj import StrictPartialOrder, OperatorPOWL, Transition, SilentTransition
from pm4py.objects.process_tree.obj import Operator

# Define all activity transitions
SR = Transition(label="SR")
CS = Transition(label="CS")
DA = Transition(label="DA")
ACI = Transition(label="ACI")
BNC = Transition(label="BNC")
DR = Transition(label="D&R")
CD = Transition(label="CD")
MD = Transition(label="MD")
QA = Transition(label="QA")
SA = Transition(label="SA")
GLA = Transition(label="GLA")

# Loop: execute BNC, then either exit or execute D&R followed by BNC again (repeats until stable)
loop = OperatorPOWL(operator=Operator.LOOP, children=[BNC, DR])

# Exclusive choice between the two deployment methods
xor = OperatorPOWL(operator=Operator.XOR, children=[CD, MD])

# Top-level partial order capturing all sequencing and concurrency
root = StrictPartialOrder(nodes=[SR, CS, DA, ACI, loop, xor, QA, SA, GLA])

# Ordering constraints
root.order.add_edge(SR, CS)          # Service Request verification before Contract Setup
root.order.add_edge(CS, DA)          # Contract Setup before Datacenter Allocation
root.order.add_edge(CS, ACI)         # Contract Setup before Access Control Initialization
root.order.add_edge(DA, loop)        # Both DA and ACI must finish before the configuration loop
root.order.add_edge(ACI, loop)
root.order.add_edge(loop, xor)       # Configuration loop before deployment choice
root.order.add_edge(xor, QA)         # Deployment before Quality Assurance
root.order.add_edge(xor, SA)         # Deployment before Security Audit
root.order.add_edge(QA, GLA)         # Both QA and SA must finish before Go-Live Approval
root.order.add_edge(SA, GLA)
