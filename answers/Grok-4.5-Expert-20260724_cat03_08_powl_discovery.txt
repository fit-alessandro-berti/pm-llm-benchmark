```python
import pm4py
from pm4py.objects.powl.obj import StrictPartialOrder, OperatorPOWL, Transition, SilentTransition
from pm4py.objects.process_tree.obj import Operator

# Basic activities
OR = Transition(label="OR")
MS = Transition(label="MS")
QS = Transition(label="QS")
AS = Transition(label="AS")
IQC = Transition(label="IQC")
TST = Transition(label="TST")
RT = Transition(label="RT")
PK = Transition(label="PK")
LB = Transition(label="LB")
DP = Transition(label="DP")
D = Transition(label="D")

# Silent transitions (distinct instances)
tau_iqc = SilentTransition()   # for zero-or-more IQC
tau_lb = SilentTransition()    # for optional Labeling

# Zero or more IQC: * (tau, IQC)
loop_iqc = OperatorPOWL(operator=Operator.LOOP, children=[tau_iqc, IQC])

# Assembly / in-line check loop: * (AS, * (tau, IQC))
# Produces traces of the form AS ( (IQC)* AS )*
assembly_loop = OperatorPOWL(operator=Operator.LOOP, children=[AS, loop_iqc])

# Testing / re-testing loop: * (TST, RT)
# Produces traces of the form TST (RT TST)*
test_loop = OperatorPOWL(operator=Operator.LOOP, children=[TST, RT])

# Optional Labeling: X (LB, tau)
xor_lb = OperatorPOWL(operator=Operator.XOR, children=[LB, tau_lb])

# Top-level partial order (sequential flow)
root = StrictPartialOrder(nodes=[OR, MS, QS, assembly_loop, test_loop, PK, xor_lb, DP, D])
root.order.add_edge(OR, MS)
root.order.add_edge(MS, QS)
root.order.add_edge(QS, assembly_loop)
root.order.add_edge(assembly_loop, test_loop)
root.order.add_edge(test_loop, PK)
root.order.add_edge(PK, xor_lb)
root.order.add_edge(xor_lb, DP)
root.order.add_edge(DP, D)
```
