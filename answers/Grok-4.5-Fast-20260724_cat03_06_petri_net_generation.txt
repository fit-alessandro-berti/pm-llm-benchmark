```python
from pm4py.objects.petri_net.obj import PetriNet, Marking
from pm4py.objects.petri_net.utils import petri_utils

# Create the Petri net
net = PetriNet(name="InsuranceClaimManagement")

# Places (states between activities)
source = PetriNet.Place("source")          # initial state: new claim arrives
p_after_C = PetriNet.Place("p_after_C")    # after Insurance Claim filed
p_after_CR = PetriNet.Place("p_after_CR")  # after Claim Registration
p_before_DV = PetriNet.Place("p_before_DV")# ready for Document Verification
p_after_DV = PetriNet.Place("p_after_DV")  # after Document Verification
p_before_FC = PetriNet.Place("p_before_FC")# ready for Fraud Check
p_after_FC = PetriNet.Place("p_after_FC")  # after Fraud Check
p_after_CA = PetriNet.Place("p_after_CA")  # after Claim Assessment
p_after_AP = PetriNet.Place("p_after_AP")  # after Approval
p_after_P = PetriNet.Place("p_after_P")    # after Payment
p_after_N = PetriNet.Place("p_after_N")    # after Notification
sink = PetriNet.Place("sink")              # final state: claim closed

# Add places to the net
for p in [source, p_after_C, p_after_CR, p_before_DV, p_after_DV,
          p_before_FC, p_after_FC, p_after_CA, p_after_AP,
          p_after_P, p_after_N, sink]:
    net.places.add(p)

# Transitions (activities)
t_C = PetriNet.Transition("C", "Insurance Claim (C)")
t_CR = PetriNet.Transition("CR", "Claim Registration (CR)")
t_DV = PetriNet.Transition("DV", "Document Verification (DV)")
t_RDS = PetriNet.Transition("RDS", "Re-Document Submission (RDS)")
t_FC = PetriNet.Transition("FC", "Fraud Check (FC)")
t_II = PetriNet.Transition("II", "In-depth Investigation (II)")
t_CA = PetriNet.Transition("CA", "Claim Assessment (CA)")
t_AP = PetriNet.Transition("AP", "Approval (AP)")
t_P = PetriNet.Transition("P", "Payment (P)")
t_N = PetriNet.Transition("N", "Notification (N)")
t_CL = PetriNet.Transition("CL", "Closure (CL)")

# Add transitions to the net
for t in [t_C, t_CR, t_DV, t_RDS, t_FC, t_II, t_CA, t_AP, t_P, t_N, t_CL]:
    net.transitions.add(t)

# Arcs – main sequential flow
petri_utils.add_arc_from_to(source, t_C, net)
petri_utils.add_arc_from_to(t_C, p_after_C, net)

petri_utils.add_arc_from_to(p_after_C, t_CR, net)
petri_utils.add_arc_from_to(t_CR, p_after_CR, net)

petri_utils.add_arc_from_to(p_after_CR, t_DV, net)          # first entry to DV
petri_utils.add_arc_from_to(p_before_DV, t_DV, net)         # subsequent entries after RDS
petri_utils.add_arc_from_to(t_DV, p_after_DV, net)

# Loop for incomplete documents: after DV → RDS → back to before DV
petri_utils.add_arc_from_to(p_after_DV, t_RDS, net)
petri_utils.add_arc_from_to(t_RDS, p_before_DV, net)

# Proceed from DV to Fraud Check
petri_utils.add_arc_from_to(p_after_DV, t_FC, net)          # first entry to FC
petri_utils.add_arc_from_to(p_before_FC, t_FC, net)         # subsequent entries after II
petri_utils.add_arc_from_to(t_FC, p_after_FC, net)

# Loop for fraud doubts: after FC → II → back to before FC (can repeat)
petri_utils.add_arc_from_to(p_after_FC, t_II, net)
petri_utils.add_arc_from_to(t_II, p_before_FC, net)

# Proceed from FC to Claim Assessment
petri_utils.add_arc_from_to(p_after_FC, t_CA, net)
petri_utils.add_arc_from_to(t_CA, p_after_CA, net)

# Continue sequential flow
petri_utils.add_arc_from_to(p_after_CA, t_AP, net)
petri_utils.add_arc_from_to(t_AP, p_after_AP, net)

petri_utils.add_arc_from_to(p_after_AP, t_P, net)
petri_utils.add_arc_from_to(t_P, p_after_P, net)

petri_utils.add_arc_from_to(p_after_P, t_N, net)
petri_utils.add_arc_from_to(t_N, p_after_N, net)

petri_utils.add_arc_from_to(p_after_N, t_CL, net)
petri_utils.add_arc_from_to(t_CL, sink, net)

# Initial marking: one token in the source place (new claim arrives)
im = Marking()
im[source] = 1

# Final marking: one token in the sink place (claim closed and archived)
fm = Marking()
fm[sink] = 1
```

The constructed accepting Petri net models the full insurance-claim process:

- Linear path: C → CR → DV → FC → CA → AP → P → N → CL  
- Optional loop after DV (incomplete documents): DV → RDS → DV (repeatable)  
- Optional loop after FC (fraud doubts): FC → II → FC (repeatable any number of times)  

Tokens flow from the initial marking (source) to the final marking (sink) only after a successful closure.
