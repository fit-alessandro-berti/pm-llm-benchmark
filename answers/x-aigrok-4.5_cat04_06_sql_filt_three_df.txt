```sql
WITH ordered_events AS (
  SELECT 
    case_id,
    activity,
    timestamp,
    LEAD(activity, 1) OVER (PARTITION BY case_id ORDER BY timestamp) AS next_activity,
    LEAD(timestamp, 1) OVER (PARTITION BY case_id ORDER BY timestamp) AS next_timestamp ,
    LEAD(activity, 2) OVER (PARTITION BY case_id ORDER BY timestamp) AS next2_activity,
    LEAD(timestamp, 2) OVER (PARTITION BY case_id ORDER BY timestamp) AS next2_timestamp
  FROM event_log
),
bad_cases AS (
  SELECT DISTINCT case_id
  FROM ordered_events
   WHERE activity = 'Approve Purchase Requisition'
     AND next_activity = 'Create Order'
     AND next2_activity = 'Confirm Order'
     AND (next2_timestamp - next_timestamp) > INTERVAL '5 days'
)
SELECT *
FROM event_log
WHERE case_id NOT IN (SELECT case FROM bad_cases);
``` 

This query works by first creating a windowed view that looks ahead two events per case, then identifies any case that contains the exact three-activity sequence with the >5-day gap between the middle and last event, and finally excludes every event belonging to those bad cases.

```sql
WITH ordered_events AS (
  SELECT
    case_id,
    activity,
    timestamp,
    LEAD(activity, 1) OVER (PARTITION BY case_id ORDER BY timestamp) AS next_activity,
    LEAD(timestamp, 1) OVER (PARTITION BY case_id ORDER BY timestamp) AS next_timestamp,
    LEAD(activity, 2) OVER (PARTITION BY case_id ORDER BY timestamp) AS next2_activity,
    LEAD(timestamp, 2) OVER (PARTITION BY case_id ORDER BY timestamp) AS next2_timestamp
  FROM event_log
),
bad_cases AS (
  SELECT DISTINCT case_id
  FROM ordered_events
  WHERE activity = 'Approve Purchase Requisition'
    AND next_activity = 'Create Order'
    AND next2_activity = 'Confirm Order'
    AND (next2_timestamp - next_timestamp) > INTERVAL '5 days'
)
SELECT *
FROM event_log
WHERE case_id NOT IN (SELECT case_id FROM bad_cases)
ORDER BY case_id, timestamp;
```
