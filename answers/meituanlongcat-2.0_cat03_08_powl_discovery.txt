LB = Transition(label="LB")      # Labeling
DP = Transition(label="DP")      # Distribution Planning
D = Transition(label="D")        # Dispatch
skip = SilentTransition()        # Represents skipping an activity

# 1. Nested Loop for IQC: Handles cases where IQC is repeated consecutively (e.g., Case 3: IQC, IQC)
# Executes IQC, then either exits or skips silently and executes IQC again.
IQC_LOOP = OperatorPOWL(operator=Operator.LOOP, children=[IQC, skip])

# 2. Loop for AS-IQC: Handles the iterative inspection and rework cycle
