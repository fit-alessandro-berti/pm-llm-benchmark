```python
from pm4py.objects.powl.obj import (
    StrictPartialOrder,
    OperatorPOWL,
    Transition
)
from pm4py.objects.process_tree.obj import Operator


# ============================================================
# 1. POWL MODEL WITH POTENTIAL UNFAIRNESS
# ============================================================

u_receive_application = Transition(label="ReceiveApplication")
u_resume_parsing = Transition(label="ResumeParsing")

# The first loop child is always executed. If information is
# incomplete, RequestMoreInfo is executed and the check repeats.
u_data_check = Transition(label="DataCompletenessCheck")
u_request_more_info = Transition(label="RequestMoreInfo")
u_data_completion_loop = OperatorPOWL(
    operator=Operator.LOOP,
    children=[u_data_check, u_request_more_info]
)

u_skill_assessment = Transition(label="SkillAssessment")

# Potentially unfair XOR:
# applicants follow either the standard evaluation or an
# affiliation-sensitive path. The latter is interpreted as
# applying a subjective uplift for qualifying affiliations.
u_cultural_fit = Transition(label="CulturalFitCheck")
u_community_affiliation = Transition(label="CommunityAffiliationCheck")
u_cultural_evaluation_xor = OperatorPOWL(
    operator=Operator.XOR,
    children=[u_cultural_fit, u_community_affiliation]
)

u_managerial_review = Transition(label="ManagerialReview")
u_final_decision = Transition(label="FinalDecision")

unfair_model = StrictPartialOrder(
    nodes=[
        u_receive_application,
        u_resume_parsing,
        u_data_completion_loop,
        u_skill_assessment,
        u_cultural_evaluation_xor,
        u_managerial_review,
        u_final_decision
    ]
)

unfair_model.order.add_edge(
    u_receive_application, u_resume_parsing
)
unfair_model.order.add_edge(
    u_resume_parsing, u_data_completion_loop
)
unfair_model.order.add_edge(
    u_data_completion_loop, u_skill_assessment
)
unfair_model.order.add_edge(
    u_skill_assessment, u_cultural_evaluation_xor
)
unfair_model.order.add_edge(
    u_cultural_evaluation_xor, u_managerial_review
)
unfair_model.order.add_edge(
    u_managerial_review, u_final_decision
)


# ============================================================
# 2. POWL MODEL WITHOUT UNFAIRNESS
# ============================================================

f_receive_application = Transition(label="ReceiveApplication")
f_resume_parsing = Transition(label="ResumeParsing")

f_data_check = Transition(label="DataCompletenessCheck")
f_request_more_info = Transition(label="RequestMoreInfo")
f_data_completion_loop = OperatorPOWL(
    operator=Operator.LOOP,
    children=[f_data_check, f_request_more_info]
)

f_skill_assessment = Transition(label="SkillAssessment")

# Every applicant undergoes exactly the same cultural evaluation.
f_cultural_fit = Transition(label="CulturalFitCheck")

f_managerial_review = Transition(label="ManagerialReview")
f_final_decision = Transition(label="FinalDecision")

fair_model = StrictPartialOrder(
    nodes=[
        f_receive_application,
        f_resume_parsing,
        f_data_completion_loop,
        f_skill_assessment,
        f_cultural_fit,
        f_managerial_review,
        f_final_decision
    ]
)

fair_model.order.add_edge(
    f_receive_application, f_resume_parsing
)
fair_model.order.add_edge(
    f_resume_parsing, f_data_completion_loop
)
fair_model.order.add_edge(
    f_data_completion_loop, f_skill_assessment
)
fair_model.order.add_edge(
    f_skill_assessment, f_cultural_fit
)
fair_model.order.add_edge(
    f_cultural_fit, f_managerial_review
)
fair_model.order.add_edge(
    f_managerial_review, f_final_decision
)
```

In `unfair_model`, the XOR between `CulturalFitCheck` and `CommunityAffiliationCheck` identifies the potential bias point. POWL represents the differing control-flow paths; the subjective score uplift is the business meaning attached to the affiliation activity.

In `fair_model`, that XOR is removed, so every applicant follows the same `CulturalFitCheck` before managerial review and the final decision.
