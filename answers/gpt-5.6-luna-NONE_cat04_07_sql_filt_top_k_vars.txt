```sql
-- Replace 10 with the desired value of K
WITH case_variants AS (
    SELECT
        case_id,
        list(activity ORDER BY timestamp) AS activity_sequence
    FROM event_log
    GROUP BY case_id
),

variant_counts AS (
    SELECT
        activity_sequence,
        COUNT(*) AS case_count
    FROM case_variants
    GROUP BY activity_sequence
),

top_k_variants AS (
    SELECT
        activity_sequence,
        case_count
    FROM variant_counts
    ORDER BY case_count DESC, activity_sequence
    LIMIT 10
),

top_k_cases AS (
    SELECT cv.case_id
    FROM case_variants AS cv
    INNER JOIN top_k_variants AS tv
        ON cv.activity_sequence = tv.activity_sequence
)

SELECT e.*
FROM event_log AS e
INNER JOIN top_k_cases AS c
    ON e.case_id = c.case_id
ORDER BY e.case_id, e.timestamp;
```

`list(activity ORDER BY timestamp)` creates the ordered activity sequence for each case. Cases with identical sequences are grouped into the same variant, counted, and ranked by frequency. The final query returns all events belonging to cases whose variant is among the top `K`.

If timestamps are not unique within a case, add a deterministic tie-breaker to the ordering, such as an event ID:

```sql
list(activity ORDER BY timestamp, event_id)
```
