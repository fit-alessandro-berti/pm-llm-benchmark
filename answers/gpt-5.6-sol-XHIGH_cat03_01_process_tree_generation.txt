The abnormal-results section must be optional because the initial diagnostic result may already be normal. If it is abnormal, the loop executes at least one specialist-treatment-retest cycle.

### Process-tree notation

```text
->(
    'A',
    'RG',
    'IA',
    ->(
        'TI',
        'TC'
    ),
    X(
        'CE',
        'FBP'
    ),
    X(
        tau,
        *(
            ->(
                'SC',
                X(
                    'TM',
                    'SPT'
                ),
                'RDT',
                X(
                    'FCE',
                    'FBA'
                )
            ),
            tau
        )
    ),
    +(
        'AI',
        'NC'
    ),
    ->(
        'FAS',
        'DDI'
    ),
    +(
        'BL',
        'ICP',
        'MRF'
    )
)
```

The outer `X(tau, ...)` after the initial test represents:

- `tau`: the initial result is normal, so no treatment loop is needed.
- `*(..., tau)`: the initial result is abnormal, so one or more consultation-treatment-retest cycles occur.

Because `RDT` is included in the expected activity set, it is modeled explicitly before the choice between `FCE` and `FBA`.

### PM4Py construction

```python
from pm4py.objects.process_tree.obj import ProcessTree, Operator


def activity(label):
    return ProcessTree(label=label)


def silent():
    return ProcessTree()  # tau: no label and no operator


def operator_node(operator, *children):
    node = ProcessTree(operator=operator)

    for child in children:
        # Explicitly establish both sides of the relationship
        child.parent = node
        node.children.append(child)

    return node


# Triage: TI before TC
triage = operator_node(
    Operator.SEQUENCE,
    activity("TI"),
    activity("TC")
)

# Exclusive initial diagnostic test
initial_test = operator_node(
    Operator.XOR,
    activity("CE"),
    activity("FBP")
)

# Exclusive treatment choice
treatment_choice = operator_node(
    Operator.XOR,
    activity("TM"),
    activity("SPT")
)

# Exclusive follow-up diagnostic choice
followup_test_choice = operator_node(
    Operator.XOR,
    activity("FCE"),
    activity("FBA")
)

# One abnormal-result treatment/retest cycle
abnormal_cycle = operator_node(
    Operator.SEQUENCE,
    activity("SC"),
    treatment_choice,
    activity("RDT"),
    followup_test_choice
)

# One or more abnormal cycles:
# abnormal_cycle, followed optionally by tau and abnormal_cycle again
abnormal_loop = operator_node(
    Operator.LOOP,
    abnormal_cycle,
    silent()
)

# Initial results may already be normal, so the loop is optional
diagnostic_resolution = operator_node(
    Operator.XOR,
    silent(),
    abnormal_loop
)

# Advanced imaging and counseling occur concurrently
advanced_support = operator_node(
    Operator.PARALLEL,
    activity("AI"),
    activity("NC")
)

# Final informational and scheduling sequence
discharge_preparation = operator_node(
    Operator.SEQUENCE,
    activity("FAS"),
    activity("DDI")
)

# Final administrative activities occur concurrently
final_clearance = operator_node(
    Operator.PARALLEL,
    activity("BL"),
    activity("ICP"),
    activity("MRF")
)

# Complete process tree
root = operator_node(
    Operator.SEQUENCE,
    activity("A"),
    activity("RG"),
    activity("IA"),
    triage,
    initial_test,
    diagnostic_resolution,
    advanced_support,
    discharge_preparation,
    final_clearance
)

# root.parent remains None, as required for the root node
```
