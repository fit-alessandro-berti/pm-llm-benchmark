Below is one normative model for successful product launches. It assumes every activity occurs exactly once, TFC and CE may occur in either order, and all listed constraints have support and confidence equal to `1.0`.

```python
def score():
    return {"support": 1.0, "confidence": 1.0}


activities = (
    "IG", "DD", "TFC", "CE", "PC",
    "LT", "UT", "AG", "MP", "FL"
)

# Required ordering. TFC and CE are both after DD and before PC,
# but no ordering is imposed between TFC and CE.
ordered_handoffs = (
    ("IG", "DD"),
    ("DD", "TFC"),
    ("DD", "CE"),
    ("TFC", "PC"),
    ("CE", "PC"),
    ("PC", "LT"),
    ("LT", "UT"),
    ("UT", "AG"),
    ("AG", "MP"),
    ("MP", "FL"),
)

# Handoffs that must be immediate in the event trace.
strict_handoffs = (
    ("IG", "DD"),
    ("PC", "LT"),
    ("LT", "UT"),
    ("UT", "AG"),
    ("AG", "MP"),
    ("MP", "FL"),
)

# Relations for which repetition would require alternation.
alternating_handoffs = (
    ("PC", "LT"),
    ("LT", "UT"),
    ("UT", "AG"),
    ("AG", "MP"),
    ("MP", "FL"),
)

responded_existence_rules = (
    ("PC", "TFC"),  # A prototype requires a feasibility check.
    ("PC", "CE"),   # A prototype requires a cost evaluation.
    ("AG", "LT"),   # Approval requires laboratory testing.
    ("AG", "UT"),   # Approval requires user testing.
    ("FL", "AG"),   # Launch requires approval.
    ("FL", "MP"),   # Launch requires a marketing plan.
)

coexistence_rules = (
    ("TFC", "CE"),
    ("LT", "UT"),
    ("AG", "MP"),
)

# Reverse ordering is forbidden.
non_succession_rules = tuple(
    (target, source) for source, target in ordered_handoffs
)

# These activities cannot occur directly next to each other because
# an intermediate process step is required.
non_chain_succession_rules = (
    ("IG", "TFC"),
    ("IG", "CE"),
    ("DD", "PC"),
    ("TFC", "LT"),
    ("CE", "LT"),
    ("PC", "UT"),
    ("LT", "AG"),
    ("UT", "MP"),
    ("AG", "FL"),
)


declare_model = {
    "existence": {
        activity: score() for activity in activities
    },

    # No activity in the defined universe is required to be absent.
    "absence": {},

    "exactly_one": {
        activity: score() for activity in activities
    },

    "init": {
        "IG": score()
    },

    "responded_existence": {
        rule: score() for rule in responded_existence_rules
    },

    "coexistence": {
        rule: score() for rule in coexistence_rules
    },

    "response": {
        rule: score() for rule in ordered_handoffs
    },

    "precedence": {
        rule: score() for rule in ordered_handoffs
    },

    "succession": {
        rule: score() for rule in ordered_handoffs
    },

    "altresponse": {
        rule: score() for rule in alternating_handoffs
    },

    "altprecedence": {
        rule: score() for rule in alternating_handoffs
    },

    "altsuccession": {
        rule: score() for rule in alternating_handoffs
    },

    "chainresponse": {
        rule: score() for rule in strict_handoffs
    },

    "chainprecedence": {
        rule: score() for rule in strict_handoffs
    },

    "chainsuccession": {
        rule: score() for rule in strict_handoffs
    },

    # There are no mutually exclusive activities in a successful launch.
    "noncoexistence": {},

    "nonsuccession": {
        rule: score() for rule in non_succession_rules
    },

    "nonchainsuccession": {
        rule: score() for rule in non_chain_succession_rules
    },
}
```

For binary constraints, a tuple `(A, B)` uses `A` as the source/activating activity and `B` as the target. For example, `("PC", "LT")` under `chainresponse` means every Prototype Creation must be immediately followed by Laboratory Testing. In an actual discovery setting, support and confidence would be calculated from an event log rather than assumed to be `1.0`.
