```sql
-- Set K to the desired number of top variants
WITH params AS (
    SELECT 5 AS top_k
),
-- 1. Variant per case: ordered sequence of activities
variants AS (
    SELECT
        case_id,
        array_agg(activity ORDER BY timestamp) AS variant_array
    FROM event_log
    GROUP BY case_id
),
-- 2. Count cases per variant
variant_stats AS (
    SELECT
        variant_array,
        COUNT(*) AS freq
    FROM variants
    GROUP BY variant_array
),
-- 3. Rank variants by frequency, keeping ties
ranked_variants AS (
    SELECT
        variant_array,
        freq,
        RANK() OVER (ORDER BY freq DESC) AS variant_rank
    FROM variant_stats
),
-- 4. Keep only variants with rank <= K
top_k_variants AS (
    SELECT variant_array
    FROM ranked_variants
    CROSS JOIN params
    WHERE variant_rank <= params.top_k
),
-- 5. Identify all cases that belong to the top K variants
valid_cases AS (
    SELECT v.case_id
    FROM variants v
    JOIN top_k_variants t ON v.variant_array = t.variant_array
)
-- 6. Return all log events for those cases
SELECT e.*
FROM event_log e
JOIN valid_cases vc ON e.case_id = vc.case_id;
```
