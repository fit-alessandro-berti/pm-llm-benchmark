```python
from pm4py.objects.powl.obj import (
    StrictPartialOrder,
    OperatorPOWL,
    Transition,
)
from pm4py.objects.process_tree.obj import Operator


def ordered(*nodes):
    """Create a transitively ordered POWL sequence."""
    model = StrictPartialOrder(nodes=list(nodes))

    # Add every implied ordering edge so the relation is transitive.
    for i, source in enumerate(nodes):
        for target in nodes[i + 1:]:
            model.order.add_edge(source, target)

    return model


def build_hiring_model(with_affiliation_branch):
    receive_application = Transition(label="ReceiveApplication")
    resume_parsing = Transition(label="ResumeParsing")
    completeness_check = Transition(label="DataCompletenessCheck")
    request_more_info = Transition(label="RequestMoreInfo")
    skill_assessment = Transition(label="SkillAssessment")
    cultural_fit_check = Transition(label="CulturalFitCheck")
    managerial_review = Transition(label="ManagerialReview")
    final_decision = Transition(label="FinalDecision")

    # Resume data is parsed and checked. If information is missing,
    # RequestMoreInfo is executed before parsing and checking it again.
    parse_and_check = ordered(
        resume_parsing,
        completeness_check,
    )

    data_completeness_loop = OperatorPOWL(
        operator=Operator.LOOP,
        children=[
            parse_and_check,
            request_more_info,
        ],
    )

    if with_affiliation_branch:
        community_affiliation_check = Transition(
            label="CommunityAffiliationCheck"
        )

        # Potentially unfair XOR:
        # affiliation-related routing can lead to a favorable score uplift.
        cultural_evaluation = OperatorPOWL(
            operator=Operator.XOR,
            children=[
                cultural_fit_check,
                community_affiliation_check,
            ],
        )
    else:
        # Every applicant receives exactly the same cultural evaluation.
        cultural_evaluation = cultural_fit_check

    return ordered(
        receive_application,
        data_completeness_loop,
        skill_assessment,
        cultural_evaluation,
        managerial_review,
        final_decision,
    )


# Model 1: contains the potentially unfair affiliation-based XOR branch.
potentially_unfair_model = build_hiring_model(
    with_affiliation_branch=True
)

# Model 2: all applicants follow the same cultural-fit process.
fair_model = build_hiring_model(
    with_affiliation_branch=False
)
```

Structurally, the models are:

```text
Potentially unfair:

ReceiveApplication
  -> LOOP(
       ResumeParsing -> DataCompletenessCheck,
       RequestMoreInfo
     )
  -> SkillAssessment
  -> XOR(CulturalFitCheck, CommunityAffiliationCheck)
  -> ManagerialReview
  -> FinalDecision
```

```text
Without unfairness:

ReceiveApplication
  -> LOOP(
       ResumeParsing -> DataCompletenessCheck,
       RequestMoreInfo
     )
  -> SkillAssessment
  -> CulturalFitCheck
  -> ManagerialReview
  -> FinalDecision
```

The score uplift itself is a business interpretation of `CommunityAffiliationCheck`; the POWL model exposes the selective branch where that treatment occurs.
