```sql
-- Bind the desired number of variants to the ? parameter.
WITH case_sequences AS (
    SELECT
        case_id,
        list(activity ORDER BY "timestamp") AS activity_sequence
    FROM event_log
    GROUP BY case_id
),

variant_counts AS (
    SELECT
        activity_sequence,
        COUNT(*) AS case_count
    FROM case_sequences
    GROUP BY activity_sequence
),

ranked_variants AS (
    SELECT
        activity_sequence,
        case_count,
        ROW_NUMBER() OVER (
            ORDER BY case_count DESC, activity_sequence
        ) AS variant_rank
    FROM variant_counts
),

top_variants AS (
    SELECT
        activity_sequence,
        case_count
    FROM ranked_variants
    WHERE variant_rank <= ?
),

top_variant_cases AS (
    SELECT
        cs.case_id
    FROM case_sequences AS cs
    INNER JOIN top_variants AS tv
        ON cs.activity_sequence = tv.activity_sequence
)

SELECT
    e.*
FROM event_log AS e
INNER JOIN top_variant_cases AS tc
    ON e.case_id = tc.case_id
ORDER BY
    e.case_id,
    e."timestamp";
```

`list(activity ORDER BY "timestamp")` creates an ordered DuckDB list for each case, which serves as the variant identifier. The query counts cases sharing each list, ranks variants by frequency, retains the top `K` variants, and finally returns only events belonging to cases with one of those variants.

If multiple events in a case can have the exact same timestamp, add a deterministic tie-breaker column to the ordering, for example:

```sql
list(activity ORDER BY "timestamp", event_id)
```

where `event_id` is a unique event-level identifier.
