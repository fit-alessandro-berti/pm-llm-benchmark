```sql
WITH case_variants AS (
  SELECT
    case_id,
    list_agg(activity ORDER BY timestamp) AS process_variant
  FROM event_log
  GROUP BY case_id
),
variant_counts AS (
  SELECT
    process_variant,
    COUNT(*) AS frequency
  FROM case_variants
  GROUP BY process_variant
  ORDER BY frequency DESC
  LIMIT K   -- replace K with the desired integer
),
top_case_ids AS (
  SELECT cv.case_id
  FROM case_variants cv
  INNER JOIN variant_counts vc
    ON cv.process_variant = vc.process_variant
)
SELECT el.*
FROM event_log el
WHERE el.case_id IN (SELECT case_id FROM top_case_ids)
ORDER BY el.case_id, el.timestamp;
```
