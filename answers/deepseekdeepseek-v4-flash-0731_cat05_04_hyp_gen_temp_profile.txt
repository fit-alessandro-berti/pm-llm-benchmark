The queries below assume PostgreSQL. The base CTE calculates the first timestamp for each activity per claim.

```sql
WITH claim_times AS (
  SELECT
    claim_id,
    MIN(timestamp) FILTER (WHERE activity = 'R') AS r_time,
    MIN(timestamp) FILTER (WHERE activity = 'A') AS a_time,
    MIN(timestamp) FILTER (WHERE activity = 'E') AS e_time,
    MIN(timestamp) FILTER (WHERE activity = 'P') AS p_time,
    MIN(timestamp) FILTER (WHERE activity = 'N') AS n_time,
    MIN(timestamp) FILTER (WHERE activity = 'C') AS c_time
  FROM claim_events
  GROUP BY claim_id
)
SELECT
  ct.claim_id,
  c.claim_type,
  EXTRACT(EPOCH FROM (ct.p_time - ct.r_time)) / 3600 AS r_to_p_hours,
  EXTRACT(EPOCH FROM (ct.n_time - ct.p_time)) / 86400 AS p_to_n_days,
  EXTRACT(EPOCH FROM (ct.c_time - ct.a_time)) / 3600 AS a_to_c_hours,
  EXTRACT(EPOCH FROM (ct.c_time - ct.e_time)) / 3600 AS e_to_c_hours,
  EXTRACT(EPOCH FROM (ct.n_time - ct.e_time)) / 60 AS e_to_n_minutes
FROM claim_times ct
JOIN claims c USING (claim_id)
WHERE
  (ct.r_time IS NOT NULL AND ct.p_time IS NOT NULL
   AND ABS(EXTRACT(EPOCH FROM (ct.p_time - ct.r_time)) - 90000) > 7200)
  OR (ct.p_time IS NOT NULL AND ct.n_time IS NOT NULL
      AND (EXTRACT(EPOCH FROM (ct.n_time - ct.p_time)) > 950400
           OR EXTRACT(EPOCH FROM (ct.n_time - ct.p_time)) < 259200))
  OR (ct.a_time IS NOT NULL AND ct.c_time IS NOT NULL
      AND EXTRACT(EPOCH FROM (ct.c_time - ct.a_time)) < 7200)
  OR (ct.e_time IS NOT NULL AND ct.c_time IS NOT NULL
      AND EXTRACT(EPOCH FROM (ct.c_time - ct.e_time)) < 3600)
  OR (ct.e_time IS NOT NULL AND ct.n_time IS NOT NULL
      AND EXTRACT(EPOCH FROM (ct.n_time - ct.e_time)) <= 300)
ORDER BY p_to_n_days DESC NULLS LAST;
```
