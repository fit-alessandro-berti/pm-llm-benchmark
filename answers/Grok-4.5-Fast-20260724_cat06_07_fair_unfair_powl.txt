```python
import pm4py
from pm4py.objects.powl.obj import StrictPartialOrder, OperatorPOWL, Transition, SilentTransition
from pm4py.objects.process_tree.obj import Operator

# ============================================================
# 1. POWL Model WITH Potential Unfairness
# ============================================================
# Activities
rec1 = Transition(label="ReceiveApplication")
check1 = Transition(label="DataCompletenessCheck")
req1 = Transition(label="RequestMoreInfo")
skill1 = Transition(label="SkillAssessment")
cult1 = Transition(label="CulturalFitCheck")
comm1 = Transition(label="CommunityAffiliationCheck")
man1 = Transition(label="ManagerialReview")
dec1 = Transition(label="FinalDecision")

# Loop for data completeness: do check, then either exit or request more info + check again
loop1 = OperatorPOWL(operator=Operator.LOOP, children=[check1, req1])

# XOR that introduces potential unfairness:
# one path = standard cultural fit, other path = community affiliation (subtle advantage)
xor_unfair = OperatorPOWL(operator=Operator.XOR, children=[cult1, comm1])

# Partial order
root_unfair = StrictPartialOrder(nodes=[rec1, loop1, skill1, xor_unfair, man1, dec1])
root_unfair.order.add_edge(rec1, loop1)
root_unfair.order.add_edge(loop1, skill1)
root_unfair.order.add_edge(skill1, xor_unfair)
root_unfair.order.add_edge(xor_unfair, man1)
root_unfair.order.add_edge(man1, dec1)

# ============================================================
# 2. POWL Model WITHOUT Unfairness
# ============================================================
# Activities (same labels where applicable)
rec2 = Transition(label="ReceiveApplication")
check2 = Transition(label="DataCompletenessCheck")
req2 = Transition(label="RequestMoreInfo")
skill2 = Transition(label="SkillAssessment")
cult2 = Transition(label="CulturalFitCheck")
man2 = Transition(label="ManagerialReview")
dec2 = Transition(label="FinalDecision")

# Same data-completeness loop
loop2 = OperatorPOWL(operator=Operator.LOOP, children=[check2, req2])

# No XOR / no CommunityAffiliationCheck – every applicant follows the same cultural-fit path
root_fair = StrictPartialOrder(nodes=[rec2, loop2, skill2, cult2, man2, dec2])
root_fair.order.add_edge(rec2, loop2)
root_fair.order.add_edge(loop2, skill2)
root_fair.order.add_edge(skill2, cult2)
root_fair.order.add_edge(cult2, man2)
root_fair.order.add_edge(man2, dec2)
```
