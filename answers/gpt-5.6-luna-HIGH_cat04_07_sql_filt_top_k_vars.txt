```sql
-- Replace 10 with the desired value of K
WITH case_variants AS (
    SELECT
        case_id,
        list(activity ORDER BY "timestamp") AS activity_sequence
    FROM event_log
    GROUP BY case_id
),

variant_counts AS (
    SELECT
        activity_sequence,
        COUNT(*) AS case_count
    FROM case_variants
    GROUP BY activity_sequence
),

top_variants AS (
    SELECT
        activity_sequence
    FROM variant_counts
    ORDER BY case_count DESC
    LIMIT 10
),

top_case_ids AS (
    SELECT cv.case_id
    FROM case_variants AS cv
    INNER JOIN top_variants AS tv
        ON cv.activity_sequence = tv.activity_sequence
)

SELECT e.*
FROM event_log AS e
INNER JOIN top_case_ids AS tc
    ON e.case_id = tc.case_id
ORDER BY
    e.case_id,
    e."timestamp";
```

`list(activity ORDER BY "timestamp")` creates the ordered activity sequence for each case. Cases with identical sequences are grouped into the same variant, and `LIMIT 10` selects the ten most frequent variants.

If multiple events in a case can have the same timestamp, an additional deterministic tie-breaker, such as `event_id`, should be included:

```sql
list(activity ORDER BY "timestamp", event_id)
```
