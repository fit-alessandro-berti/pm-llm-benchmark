```python
from pm4py.objects.powl.obj import (
    StrictPartialOrder,
    OperatorPOWL,
    Transition,
    SilentTransition,
)
from pm4py.objects.process_tree.obj import Operator


# Activities
sr = Transition(label="Service Request (SR)")
cs = Transition(label="Contract Setup (CS)")

da = Transition(label="Datacenter Allocation (DA)")
aci = Transition(label="Access Control Initialization (ACI)")

bnc = Transition(label="Base Network Configuration (BNC)")
debug_reconfigure = Transition(label="Debug & Reconfigure (D&R)")

cd = Transition(label="Containerized Deployment (CD)")
md = Transition(label="Monolithic Deployment (MD)")

qa = Transition(label="Quality Assurance (QA)")
sa = Transition(label="Security Audit (SA)")
gla = Transition(label="Go-Live Approval (GLA)")


# Loop:
# Execute BNC, then either exit or perform D&R and repeat BNC.
configuration_loop = OperatorPOWL(
    operator=Operator.LOOP,
    children=[bnc, debug_reconfigure],
)


# Exclusive choice between deployment methods.
deployment_choice = OperatorPOWL(
    operator=Operator.XOR,
    children=[cd, md],
)


# Root partial order.
# The nodes DA and ACI are intentionally not ordered with respect to each
# other, so they can execute concurrently.
root = StrictPartialOrder(
    nodes=[
        sr,
        cs,
        da,
        aci,
        configuration_loop,
        deployment_choice,
        qa,
        sa,
        gla,
    ]
)


# Initial request verification precedes contract setup.
root.order.add_edge(sr, cs)

# Contract setup must complete before either environment-preparation task.
root.order.add_edge(cs, da)
root.order.add_edge(cs, aci)

# Both preparation tasks must complete before configuration begins.
root.order.add_edge(da, configuration_loop)
root.order.add_edge(aci, configuration_loop)

# Configuration must complete before choosing a deployment method.
root.order.add_edge(configuration_loop, deployment_choice)

# The selected deployment must complete before QA and the security audit.
root.order.add_edge(deployment_choice, qa)
root.order.add_edge(deployment_choice, sa)

# QA and SA are concurrent, but both must complete before go-live approval.
root.order.add_edge(qa, gla)
root.order.add_edge(sa, gla)
```

Here, the loop executes **BNC** first and may repeatedly execute **D&R** followed by another **BNC** before exiting. The deployment choice is exclusive, while **DA/ACI** and **QA/SA** remain unordered and can execute concurrently.
