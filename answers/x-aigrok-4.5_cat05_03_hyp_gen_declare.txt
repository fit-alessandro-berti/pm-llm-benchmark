/* 3.4 Temporal order check: C appears before R (violates precedence) or C never preceded by R */
WITH ordered AS (
  SELECT claim_id, activity, timestamp,
         MIN(timestamp) FILTER (WHERE activity = 'R') OVER (PARTITION BY claim_id) AS first_R,
         MIN(timestamp) FILTER (WHERE activity = 'C') OVER (PARTITION BY claim_id) AS first_C
  FROM   claim_events
)
SELECT DISTINCT claim_id
FROM   ordered
WHERE  first_C IS NOT NULL
  AND  (first_R IS NULL OR first_C < first_R);

/* 3.5 Completeness of intended path – any of the six mandatory activities missing */
SELECT c.claim_id
FROM   claims c
WHERE  NOT (

     AND EXISTS (SELECT 1 FROM claim_events WHERE claim_id = c.claim_id AND activity = 'P')
     AND EXISTS (SELECT 1 FROM claim_events WHERE claim_id = c.claim_id AND activity = 'N')
     AND EXISTS (SELECT 1 FROM claim_events WHERE claim_id = c.claim_id AND activity = 'C')
);

/* 3.6 Resource / specialisation mismatch for evaluation events */
SELECT ce.claim_id, ce.timestamp, ce.resource, a.specialization, c.claim_type
FROM   claim_events ce
JOIN   claims c    ON c.claim_id = ce.claim_id

/* 3.7 Frequency of every activity (to quantify how often E is absent despite existence(C)) */
SELECT activity, COUNT(DISTINCT claim_id) AS cases_with_activity
FROM   claim_events
GROUP  BY activity
ORDER  BY cases_with_activity DESC;
